local function getSafeParent()
    if type(gethui) == "function" then
        local ok, ui = pcall(gethui)
        if ok and ui then return ui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then return cg end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function make(className, props, children)
    local inst = Instance.new(className)
    if props then
        for k, v in pairs(props) do
            if k ~= "Parent" then
                inst[k] = v
            end
        end
    end
    if children then
        for _, c in ipairs(children) do
            c.Parent = inst
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function clamp(x, lo, hi)
    if x < lo then return lo end
    if x > hi then return hi end
    return x
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

local function tween(obj, duration, props, style, dir)
    local t = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local function attachCleanPress(button, targetFrame, flashFrame, scaleAmount)
    local uiScale = targetFrame:FindFirstChildOfClass("UIScale") or make("UIScale", {
        Scale = 1,
        Parent = targetFrame,
    })
    local pressScale = scaleAmount or 0.98

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(uiScale, 0.08, { Scale = pressScale }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            if flashFrame then
                flashFrame.BackgroundColor3 = Palette.GlassRowFlash
                tween(flashFrame, 0.08, { BackgroundTransparency = 0.56 })
            end
        end
    end)

    local function release()
        tween(uiScale, 0.20, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        if flashFrame then
            tween(flashFrame, 0.22, { BackgroundTransparency = 1, BackgroundColor3 = Palette.GlassRowHover })
        end
    end

    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            release()
        end
    end)

    button.MouseLeave:Connect(release)
end

local Maid = {}
Maid.__index = Maid

function Maid.new()
    return setmetatable({ _items = {} }, Maid)
end

function Maid:Give(item)
    table.insert(self._items, item)
    return item
end

function Maid:Clean()
    for i = #self._items, 1, -1 do
        local item = self._items[i]
        self._items[i] = nil
        local t = typeof(item)
        if t == "RBXScriptConnection" then
            item:Disconnect()
        elseif t == "Instance" then
            item:Destroy()
        elseif t == "function" then
            pcall(item)
        end
    end
end

local Spring = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping)
    return setmetatable({
        Position = initial or 0,
        Target = initial or 0,
        Velocity = 0,
        Stiffness = stiffness or 280,
        Damping = damping or 22,
    }, Spring)
end

function Spring:Step(dt)
    local d = math.min(dt, 1 / 30)
    local force = (self.Target - self.Position) * self.Stiffness
    self.Velocity = self.Velocity + (force - self.Velocity * self.Damping) * d
    self.Position = self.Position + self.Velocity * d
    return self.Position
end

local SilkCloth = {}
SilkCloth.__index = SilkCloth

local COLS = 16
local ROWS = 10
local SPAN_X = 16.8
local SPAN_Y = 11.4
local MAX_PULSES = 3

local LIGHT_KEY = Vector3.new(-0.42, 0.56, 0.71).Unit
local LIGHT_RIM = Vector3.new(0.68, -0.34, 0.65).Unit
local VIEW_DIR  = Vector3.new(0, 0, 1)
local HALF_KEY  = (LIGHT_KEY + VIEW_DIR).Unit
local HALF_RIM  = (LIGHT_RIM + VIEW_DIR).Unit

local function shadeSilk(normal, height, touchGlow)
    local nDotL = normal:Dot(LIGHT_KEY)
    local wrap = clamp((nDotL + 0.26) / 1.26, 0, 1)

    local nDotH = math.max(0, normal:Dot(HALF_KEY))
    local satinSheen = math.pow(nDotH, 5.8) * 0.60
    local sharpGlint = math.pow(nDotH, 20) * 0.54

    local rimDotH = math.max(0, normal:Dot(HALF_RIM))
    local rimSheen = math.pow(rimDotH, 8.5) * 0.28

    local fresnel = math.pow(1 - clamp(math.abs(normal.Z), 0, 1), 1.95) * 0.46
    local foldDepth = clamp((height + 1.05) / 2.1, 0, 1)

    local lum = clamp(wrap * 0.56 + foldDepth * 0.22 + fresnel + touchGlow * 0.28, 0, 1)
    local spec = clamp(satinSheen + sharpGlint + rimSheen + touchGlow * 0.34, 0, 1)

    -- Deepened +5% crimson saturation & darker obsidian valleys
    local r = clamp(0.018 + lum * 0.56 + spec * 0.50, 0, 0.99)
    local g = clamp(0.006 + lum * 0.032 + math.pow(spec, 1.85) * 0.34, 0, 0.52)
    local b = clamp(0.012 + lum * 0.068 + math.pow(spec, 1.65) * 0.38, 0, 0.58)
    return Color3.new(r, g, b)
end

local function updateTriangle(w1, w2, a, b, c, color)
    local ab = (b - a).Magnitude
    local ac = (c - a).Magnitude
    local bc = (c - b).Magnitude

    if ac > ab and ac > bc then
        b, c = c, b
        ab = ac
    elseif bc > ab and bc > ac then
        a, c = c, a
        ab = bc
    end

    if ab < 1e-4 then return end
    local dir = (b - a) / ab
    local acVec = c - a
    local proj = acVec:Dot(dir)
    local d = a + dir * proj
    local alt = c - d
    local h = alt.Magnitude
    if h < 1e-4 then return end

    local up = alt / h
    local back = dir:Cross(up)
    if back.Magnitude < 1e-4 then return end
    back = back.Unit

    -- Slight seam overlap (+0.018) eliminates hairline sub-pixel cracking
    local len1 = math.max(proj, 0.01) + 0.018
    local len2 = math.max(ab - proj, 0.01) + 0.018
    local hh = h + 0.018

    w1.Size = Vector3.new(0.11, hh, len1)
    w1.CFrame = CFrame.fromMatrix((a + c) * 0.5, -back, up, dir)
    w1.Color = color

    w2.Size = Vector3.new(0.11, hh, len2)
    w2.CFrame = CFrame.fromMatrix((b + c) * 0.5, back, up, -dir)
    w2.Color = color
end

function SilkCloth.new(hostFrame, maid, cornerRadius)
    local self = setmetatable({
        Host = hostFrame,
        Maid = maid,
        Time = 0,
        Active = true,
        PointerTarget = Vector2.new(0, 0),
        PointerCurrent = Vector2.new(0, 0),
        PointerVelocity = Vector2.new(0, 0),
        PointerEnergy = 0,
        PointerInside = false,
        PointerDown = false,
        Pulses = {},
        PulseCursor = 1,
        Vertices = {},
        Triangles = {},
    }, SilkCloth)

    for i = 1, MAX_PULSES do
        self.Pulses[i] = { x = 0, y = 0, age = 10, strength = 0 }
    end

    local viewport = make("ViewportFrame", {
        Name = "SilkViewport",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BorderSizePixel = 0,
        Ambient = Color3.new(1, 1, 1),
        LightColor = Color3.fromRGB(34, 14, 20),
        LightDirection = Vector3.new(-0.3, -0.5, -0.8),
        ZIndex = 1,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
    })

    local cam = make("Camera", {
        FieldOfView = 32,
        CFrame = CFrame.lookAt(Vector3.new(0, 0, 10.2), Vector3.new(0, 0, 0)),
        Parent = viewport,
    })
    viewport.CurrentCamera = cam

    local meshFolder = make("Folder", { Name = "ClothMesh", Parent = viewport })

    local vCount = (COLS + 1) * (ROWS + 1)
    for i = 1, vCount do
        self.Vertices[i] = Vector3.zero
    end

    local function newWedge()
        return make("WedgePart", {
            Anchored = true,
            CanCollide = false,
            CastShadow = false,
            Material = Enum.Material.SmoothPlastic,
            Color = Palette.Base,
            Size = Vector3.new(0.11, 0.1, 0.1),
            Parent = meshFolder,
        })
    end

    for row = 0, ROWS - 1 do
        for col = 0, COLS - 1 do
            local i00 = row * (COLS + 1) + col + 1
            local i10 = i00 + 1
            local i01 = i00 + (COLS + 1)
            local i11 = i01 + 1

            table.insert(self.Triangles, { i00, i10, i11, newWedge(), newWedge() })
            table.insert(self.Triangles, { i00, i11, i01, newWedge(), newWedge() })
        end
    end

    -- Deeper scrim (+5% richness) for contrast and crisp legibility
    make("Frame", {
        Name = "SilkScrim",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BackgroundTransparency = 0.30,
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
        make("UIGradient", {
            Rotation = 90,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 0.26),
                NumberSequenceKeypoint.new(0.45, 0.54),
                NumberSequenceKeypoint.new(1.00, 0.18),
            }),
        }),
    })

    self:_bindInput()
    self:_startLoop()
    return self
end

function SilkCloth:_screenToCloth(screenPos)
    local absPos = self.Host.AbsolutePosition
    local absSize = self.Host.AbsoluteSize
    if absSize.X <= 1 or absSize.Y <= 1 then
        return 0, 0, false
    end
    local nx = ((screenPos.X - absPos.X) / absSize.X) * 2 - 1
    local ny = 1 - ((screenPos.Y - absPos.Y) / absSize.Y) * 2
    local inside = nx >= -1 and nx <= 1 and ny >= -1 and ny <= 1
    return clamp(nx, -1, 1) * 4.7, clamp(ny, -1, 1) * 3.15, inside
end

function SilkCloth:TriggerPulse(screenPos, strength)
    local cx, cy = 0, 0
    if screenPos then
        cx, cy = self:_screenToCloth(screenPos)
    end
    local slot = self.Pulses[self.PulseCursor]
    slot.x = cx
    slot.y = cy
    slot.age = 0
    slot.strength = strength or 1.15
    self.PulseCursor = (self.PulseCursor % MAX_PULSES) + 1
end

function SilkCloth:_bindInput()
    self.Maid:Give(UserInputService.InputChanged:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            self.PointerInside = inside
            if inside then
                self.PointerTarget = Vector2.new(cx, cy)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputBegan:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            if inside then
                self.PointerInside = true
                self.PointerDown = true
                self.PointerTarget = Vector2.new(cx, cy)
                self:TriggerPulse(input.Position, 1.25)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            self.PointerDown = false
        end
    end))
end

function SilkCloth:_startLoop()
    self.Maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not self.Active or not self.Host.Visible then return end
        local step = math.min(dt, 0.05)
        self.Time = self.Time + step
        local t = self.Time

        local prevPtr = self.PointerCurrent
        self.PointerCurrent = self.PointerCurrent:Lerp(self.PointerTarget, 1 - math.exp(-step * 18))
        local vel = (self.PointerCurrent - prevPtr) / math.max(step, 1 / 120)
        self.PointerVelocity = self.PointerVelocity:Lerp(vel, 1 - math.exp(-step * 10))

        local targetEnergy = self.PointerDown and 1.65 or (self.PointerInside and 0.92 or 0)
        self.PointerEnergy = lerp(self.PointerEnergy, targetEnergy, 1 - math.exp(-step * 9))

        for i = 1, MAX_PULSES do
            local p = self.Pulses[i]
            if p.strength > 0 then
                p.age = p.age + step
                if p.age > 1.75 then p.strength = 0 end
            end
        end

        local px, py = self.PointerCurrent.X, self.PointerCurrent.Y
        local dragBoost = math.min(self.PointerVelocity.Magnitude * 0.065, 1.05)
        local energy = self.PointerEnergy

        -- 1. Evaluate high-density wind-blown cloth vertices with boosted touch reaction
        local verts = self.Vertices
        local idx = 1
        for row = 0, ROWS do
            local baseY = (row / ROWS - 0.5) * SPAN_Y
            for col = 0, COLS do
                local baseX = (col / COLS - 0.5) * SPAN_X

                local w1 = math.sin(baseX * 0.50 + baseY * 0.35 - t * 1.25) * 0.60
                local w2 = math.sin(baseX * 0.96 - baseY * 0.64 - t * 1.82 + w1 * 0.92) * 0.30
                local w3 = math.cos(baseX * 1.68 + baseY * 1.24 - t * 2.48 + w2 * 0.65) * 0.13

                local swayX = math.sin(baseY * 0.52 - t * 0.96) * 0.14
                local swayY = math.cos(baseX * 0.44 - t * 0.80) * 0.10

                local z = w1 + w2 + w3

                if energy > 0.01 then
                    local dx = baseX - px
                    local dy = baseY - py
                    local r2 = dx * dx + dy * dy
                    local dent = -math.exp(-r2 / 1.75) * (1.08 + dragBoost * 0.68) * energy
                    local r = math.sqrt(r2)
                    local collar = math.exp(-math.pow(r - 1.65, 2) / 0.85) * 0.38 * energy
                    z = z + dent + collar
                end

                for pIdx = 1, MAX_PULSES do
                    local pulse = self.Pulses[pIdx]
                    if pulse.strength > 0 then
                        local dx = baseX - pulse.x
                        local dy = baseY - pulse.y
                        local dist = math.sqrt(dx * dx + dy * dy)
                        local front = (dist - pulse.age * 6.4) / 0.98
                        local env = math.exp(-(front * front)) * pulse.strength * math.exp(-pulse.age * 1.55)
                        z = z + math.cos(front * 2.7) * env * 0.82
                    end
                end

                verts[idx] = Vector3.new(baseX + swayX, baseY + swayY, z)
                idx = idx + 1
            end
        end

        -- 2. Shade and update all 320 reflective triangles
        for _, tri in ipairs(self.Triangles) do
            local a = verts[tri[1]]
            local b = verts[tri[2]]
            local c = verts[tri[3]]

            local normal = (b - a):Cross(c - a)
            local nLen = normal.Magnitude
            if nLen > 1e-5 then
                normal = normal / nLen
                if normal.Z < 0 then normal = -normal end
            else
                normal = VIEW_DIR
            end

            local centroid = (a + b + c) / 3
            local touchGlow = 0
            if energy > 0.01 then
                local dx = centroid.X - px
                local dy = centroid.Y - py
                touchGlow = math.exp(-(dx * dx + dy * dy) / 3.0) * energy
            end

            local color = shadeSilk(normal, centroid.Z, touchGlow)
            updateTriangle(tri[4], tri[5], a, b, c, color)
        end
    end))
end

local function attachIrregularCrimsonBorder(targetFrame, maid, thickness)
    local stroke = make("UIStroke", {
        Color = Color3.new(1, 1, 1),
        Thickness = thickness or 1.8,
        Transparency = 0.08,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = targetFrame,
    })

    local grad = make("UIGradient", {
        Rotation = 0,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.Base),
            ColorSequenceKeypoint.new(0.28, Palette.Oxblood),
            ColorSequenceKeypoint.new(0.50, Palette.Scarlet),
            ColorSequenceKeypoint.new(0.66, Palette.RoseGlint),
            ColorSequenceKeypoint.new(0.84, Palette.Oxblood),
            ColorSequenceKeypoint.new(1.00, Palette.Base),
        }),
        Parent = stroke,
    })

    local phase = math.random() * 10
    local angle = math.random(0, 360)
    local surge = 0

    maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not targetFrame.Visible then return end
        local step = math.min(dt, 0.05)
        phase = phase + step
        surge = surge * math.exp(-step * 4.0)

        local chaoticSpeed = 26
            + 58 * math.sin(phase * 1.4142) * math.cos(phase * 0.6180)
            + 42 * math.sin(phase * 2.7183 + math.cos(phase * 1.732))
            + surge * 280

        angle = (angle + chaoticSpeed * step) % 360
        grad.Rotation = angle

        local p1 = clamp(0.22 + 0.13 * math.sin(phase * 1.618 + 0.7), 0.08, 0.34)
        local p2 = clamp(0.49 + 0.14 * math.cos(phase * 2.236 - 0.4), p1 + 0.08, 0.65)
        local p3 = clamp(p2 + 0.10 + 0.05 * math.sin(phase * 3.1415), p2 + 0.05, 0.80)
        local p4 = clamp(p3 + 0.10, p3 + 0.05, 0.93)

        grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.Base),
            ColorSequenceKeypoint.new(p1, Palette.Oxblood),
            ColorSequenceKeypoint.new(p2, Palette.Crimson),
            ColorSequenceKeypoint.new(p3, Palette.RoseGlint),
            ColorSequenceKeypoint.new(p4, Palette.Oxblood),
            ColorSequenceKeypoint.new(1.00, Palette.Base),
        })
    end))

    return function(amount)
        surge = math.min(surge + (amount or 1.0), 2.4)
    end
end

-- Water-wave ripple confined inside fixed-width waveLayer (never stretches dock!)
local function spawnTabWaterWave(waveContainer, originScreenPos)
    local absPos = waveContainer.AbsolutePosition
    local absSize = waveContainer.AbsoluteSize
    local rx = originScreenPos and (originScreenPos.X - absPos.X) or (absSize.X * 0.5)
    local ry = originScreenPos and (originScreenPos.Y - absPos.Y) or (absSize.Y * 0.5)
    local maxW = math.max(absSize.X * 1.35, 160)

    for ringIdx = 1, 2 do
        task.delay((ringIdx - 1) * 0.08, function()
            if not waveContainer or not waveContainer.Parent then return end
            local ring = make("Frame", {
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.fromOffset(rx, ry),
                Size = UDim2.fromOffset(10, 10),
                BackgroundTransparency = 1,
                BorderSizePixel = 0,
                ZIndex = 22,
                Parent = waveContainer,
            }, {
                make("UICorner", { CornerRadius = UDim.new(1, 0) }),
            })

            local st = make("UIStroke", {
                Color = (ringIdx == 1) and Palette.WaterRipple or Palette.RoseGlint,
                Thickness = (ringIdx == 1) and 2.2 or 1.4,
                Transparency = (ringIdx == 1) and 0.18 or 0.40,
                Parent = ring,
            })

            local dur = (ringIdx == 1) and 0.48 or 0.60
            tween(st, dur, { Thickness = 0.4, Transparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            local tw = tween(ring, dur, {
                Size = UDim2.fromOffset(maxW, maxW * 0.52),
            }, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out)
            tw.Completed:Once(function()
                if ring then ring:Destroy() end
            end)
        end)
    end
end

local Velour = {
    Version = "5.0.0",
    Flags = {},
}
Velour.__index = Velour

local function attachDrag(handle, target, onTap)
    local dragging = false
    local dragStart, startPos
    local moved = false
    local conns = {}

    table.insert(conns, handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))

    table.insert(conns, UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                moved = true
            end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end
    end))

    table.insert(conns, UserInputService.InputEnded:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            if not moved and onTap then
                onTap(input.Position)
            end
        end
    end))

    return function()
        for _, c in ipairs(conns) do
            c:Disconnect()
        end
    end
end

function Velour:CreateWindow(options)

    function window:SetVisible(state)
        if self.Destroyed then return end
        self.Visible = state
        if state then
            main.Visible = true
            silk.Active = true
            mainScale.Scale = 0.90
            tween(mainScale, 0.28, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            silk:TriggerPulse(main.AbsolutePosition + main.AbsoluteSize * 0.5, 1.15)
        else
            silk.Active = false
            local tw = tween(mainScale, 0.16, { Scale = 0.88 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            tw.Completed:Once(function()
                if not self.Visible and not self.Destroyed then
                    main.Visible = false
                end
            end)
        end
    end

    function window:Destroy()
        if self.Destroyed then return end
        self.Destroyed = true
        silk.Active = false
        maid:Clean()
        if env._VelourActiveDestroy == self._destroyRef then
            env._VelourActiveDestroy = nil
        end
    end
    window._destroyRef = function() window:Destroy() end
    env._VelourActiveDestroy = window._destroyRef

    function window:AddTab(tabOptions)
        tabOptions = type(tabOptions) == "string" and { Title = tabOptions } or (tabOptions or {})
        local tabIndex = #self.Tabs + 1
        local tabTitle = tabOptions.Title or ("Tab " .. tabIndex)

        local btn = make("TextButton", {
            Name = "Tab_" .. tabTitle,
            Size = UDim2.new(0, 0, 1, 0),
            AutomaticSize = Enum.AutomaticSize.X,
            BackgroundTransparency = 1,
            Text = tabTitle,
            TextColor3 = Palette.TextSecondary,
            TextSize = 12,
            FontFace = Fonts.Semi,
            AutoButtonColor = false,
            LayoutOrder = tabIndex,
            ZIndex = 24,
            Parent = dockList,
        }, {
            make("UIPadding", { PaddingLeft = UDim.new(0, 16), PaddingRight = UDim.new(0, 16) }),
        })

        local btnScale = make("UIScale", { Scale = 1, Parent = btn })

        -- Scrollbar synced to crimson-silk theme
        local page = make("ScrollingFrame", {
            Name = "Page_" .. tabTitle,
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ScrollBarThickness = 3,
            ScrollBarImageColor3 = Palette.Scarlet,
            ScrollBarImageTransparency = 0.22,
            CanvasSize = UDim2.fromOffset(0, 0),
            AutomaticCanvasSize = Enum.AutomaticSize.Y,
            ScrollingEnabled = true,
            Visible = false,
            ZIndex = 15,
            Parent = pageHolder,
        }, {
            make("UIListLayout", {
                Padding = UDim.new(0, 16),
                SortOrder = Enum.SortOrder.LayoutOrder,
            }),
            make("UIPadding", {
                PaddingTop = UDim.new(0, 4),
                PaddingBottom = UDim.new(0, 14),
                PaddingRight = UDim.new(0, 8),
            }),
        })

        local tab = {
            Index = tabIndex,
            Title = tabTitle,
            Button = btn,
            Page = page,
            Order = 0,
        }

        local function selectTab()
            if self.ActiveTab == tab then return end
            local prevIndex = self.ActiveTab and self.ActiveTab.Index or 0
            local dir = (tab.Index >= prevIndex) and 1 or -1

            for _, t in ipairs(self.Tabs) do
                t.Page.Visible = false
                tween(t.Button, 0.18, { TextColor3 = Palette.TextSecondary })
            end

            self.ActiveTab = tab
            tab.Page.Visible = true
            tween(tab.Button, 0.18, { TextColor3 = Palette.TextPrimary })

            -- Directional elastic inertia slide following the tab navigation direction
            if prevIndex > 0 then
                tab.Page.Position = UDim2.fromOffset(dir * 42, 0)
                tween(tab.Page, 0.34, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            else
                tab.Page.Position = UDim2.fromOffset(0, 0)
            end
        end

        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(btnScale, 0.08, { Scale = 0.91 })
                spawnTabWaterWave(dockWaveLayer, input.Position)
                pulseDockBorder(1.3)
            end
        end)

        btn.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(btnScale, 0.22, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            end
        end)

        btn.MouseButton1Click:Connect(function()
            selectTab()
            silk:TriggerPulse(btn.AbsolutePosition + btn.AbsoluteSize * 0.5, 1.05)
        end)

        table.insert(self.Tabs, tab)
        task.defer(syncDockWidth)

        if #self.Tabs == 1 then
            selectTab()
        end

            function section:AddToggle(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Toggle"
                local desc = cfg.Description
                local flag = cfg.Flag
                local value = cfg.Default == true
                local callback = cfg.Callback

                local row = addRow(desc and 60 or 46)

                local hit = make("TextButton", {
                    Size = UDim2.fromScale(1, 1),
                    BackgroundTransparency = 1,
                    Text = "",
                    AutoButtonColor = false,
                    ZIndex = 18,
                    Parent = row,
                })

                attachCleanPress(hit, row, row, 0.988)

                make("TextLabel", {
                    Position = UDim2.new(0, 15, 0, desc and 10 or 0),
                    Size = UDim2.new(1, -72, 0, desc and 18 or 46),
                    BackgroundTransparency = 1,
                    Text = label,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 13,
                    FontFace = Fonts.Semi,
                    TextTruncate = Enum.TextTruncate.AtEnd,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 19,
                    Parent = row,
                })

                if desc then
                    make("TextLabel", {
                        Position = UDim2.new(0, 15, 0, 31),
                        Size = UDim2.new(1, -72, 0, 20),
                        BackgroundTransparency = 1,
                        Text = desc,
                        TextColor3 = Palette.TextSecondary,
                        TextSize = 11,
                        LineHeight = 1.25,
                        FontFace = Fonts.Reg,
                        TextTruncate = Enum.TextTruncate.AtEnd,
                        TextXAlignment = Enum.TextXAlignment.Left,
                        ZIndex = 19,
                        Parent = row,
                    })
                end

                local track = make("Frame", {
                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -15, 0.5, 0),
                    Size = UDim2.fromOffset(38, 21),
                    BackgroundColor3 = Palette.Hairline,
                    BackgroundTransparency = 0.84,
                    BorderSizePixel = 0,
                    ZIndex = 19,
                    Parent = row,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local knob = make("Frame", {
                    AnchorPoint = Vector2.new(0, 0.5),
                    Position = UDim2.new(0, 3, 0.5, 0),
                    Size = UDim2.fromOffset(15, 15),
                    BackgroundColor3 = Palette.TextSecondary,
                    BorderSizePixel = 0,
                    ZIndex = 20,
                    Parent = track,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local obj = {}

                function obj:Set(v, silent)
                    value = v == true
                    if flag then Velour.Flags[flag] = value end

                    tween(knob, 0.09, { Size = UDim2.fromOffset(19, 15) }, Enum.EasingStyle.Quad)
                    task.delay(0.09, function()
                        if knob and knob.Parent then
                            tween(knob, 0.16, { Size = UDim2.fromOffset(15, 15) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        end
                    end)

                    if value then
                        tween(track, 0.20, { BackgroundColor3 = Palette.Accent, BackgroundTransparency = 0.05 })
                        tween(knob, 0.22, {
                            Position = UDim2.new(1, -18, 0.5, 0),
                            BackgroundColor3 = Palette.TextPrimary,
                        }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    else
                        tween(track, 0.20, { BackgroundColor3 = Palette.Hairline, BackgroundTransparency = 0.84 })
                        tween(knob, 0.20, {
                            Position = UDim2.new(0, 3, 0.5, 0),
                            BackgroundColor3 = Palette.TextSecondary,
                        })
                    end

                    if not silent and callback then
                        task.spawn(callback, value)
                    end
                end

                function obj:Get() return value end

                hit.MouseButton1Click:Connect(function()
                    obj:Set(not value)
                    silk:TriggerPulse(track.AbsolutePosition + track.AbsoluteSize * 0.5, 0.95)
                end)

            function section:AddSlider(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Slider"
                local minV = cfg.Min or 0
                local maxV = cfg.Max or 100
                local step = cfg.Increment or 1
                local suffix = cfg.Suffix or ""
                local flag = cfg.Flag
                local value = clamp(cfg.Default or minV, minV, maxV)
                local callback = cfg.Callback

                local row = addRow(62)

                make("TextLabel", {
                    Position = UDim2.new(0, 15, 0, 10),
                    Size = UDim2.new(1, -96, 0, 18),
                    BackgroundTransparency = 1,

                    TextColor3 = Palette.TextPrimary,
                    TextSize = 13,
                    FontFace = Fonts.Semi,

                    ZIndex = 18,

                local readout = make("TextLabel", {

                local hit = make("TextButton", {

                    BackgroundTransparency = 1,
                    Text = "",
                    AutoButtonColor = false,
                    ZIndex = 19,
                    Parent = row,
                })

                local rail = make("Frame", {
                    AnchorPoint = Vector2.new(0, 0.5),
                    Position = UDim2.fromScale(0, 0.5),
                    Size = UDim2.new(1, 0, 0, 4),
                    BackgroundColor3 = Palette.Hairline,
                    BackgroundTransparency = 0.84,
                    BorderSizePixel = 0,
                    ZIndex = 18,
                    Parent = hit,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local fill = make("Frame", {
                    Size = UDim2.fromScale(0, 1),
                    BackgroundColor3 = Palette.Accent,
                    BorderSizePixel = 0,
                    ZIndex = 19,
                    Parent = rail,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                    make("UIGradient", {
                        Color = ColorSequence.new({
                            ColorSequenceKeypoint.new(0, Palette.Oxblood),
                            ColorSequenceKeypoint.new(0.65, Palette.Crimson),
                            ColorSequenceKeypoint.new(1, Palette.RoseGlint),
                        }),
                    }),
                })

                local thumb = make("Frame", {
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromScale(1, 0.5),
                    Size = UDim2.fromOffset(13, 13),
                    BackgroundColor3 = Palette.TextPrimary,
                    BorderSizePixel = 0,
                    ZIndex = 20,
                    Parent = fill,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                    make("UIStroke", { Color = Palette.Crimson, Thickness = 1.6, Transparency = 0.2 }),
                })

                local obj = {}
                local activeInput = nil

                local function fmt(n)
                    if step < 1 then
                        local d = math.max(1, math.ceil(-math.log10(step)))
                        return string.format("%." .. d .. "f", n)
                    end
                    return tostring(math.floor(n + 0.5))
                end

                function obj:Set(v, silent)
                    local snapped = math.floor(((v - minV) / step) + 0.5) * step + minV
                    value = clamp(snapped, minV, maxV)
                    if flag then Velour.Flags[flag] = value end
                    local pct = (maxV == minV) and 0 or (value - minV) / (maxV - minV)
                    readout.Text = fmt(value) .. suffix
                    tween(fill, 0.05, { Size = UDim2.fromScale(pct, 1) })
                    if not silent and callback then
                        task.spawn(callback, value)
                    end
                end

                function obj:Get() return value end

                local function updateX(x)
                    local rel = (x - rail.AbsolutePosition.X) / math.max(rail.AbsoluteSize.X, 1)
                    obj:Set(minV + clamp(rel, 0, 1) * (maxV - minV))
                end

                local function endDrag()
                    if window._activeSlider ~= obj then return end
                    window._activeSlider = nil
                    activeInput = nil
                    page.ScrollingEnabled = true
                    tween(thumb, 0.14, { Size = UDim2.fromOffset(13, 13) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    tween(readout, 0.14, { TextColor3 = Palette.TextSection })
                    tween(readoutScale, 0.14, { Scale = 1 })
                end

                hit.InputBegan:Connect(function(input)
                    if window._activeSlider ~= nil then return end
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        window._activeSlider = obj
                        activeInput = input
                        page.ScrollingEnabled = false
                        tween(thumb, 0.12, { Size = UDim2.fromOffset(16, 16) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        tween(readout, 0.12, { TextColor3 = Palette.AccentBright })
                        tween(readoutScale, 0.12, { Scale = 1.07 })
                        updateX(input.Position.X)
                        silk:TriggerPulse(input.Position, 0.85)
                    end
                end)

                maid:Give(UserInputService.InputChanged:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement
                        or (input.UserInputType == Enum.UserInputType.Touch and input == activeInput) then
                        updateX(input.Position.X)
                    end
                end))

                maid:Give(UserInputService.InputEnded:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input == activeInput
                        or (activeInput and activeInput.UserInputType == Enum.UserInputType.MouseButton1
                            and input.UserInputType == Enum.UserInputType.MouseButton1) then
                        endDrag()
                    end
                end))

                obj:Set(value, true)
                return obj
            end

            function section:AddDropdown(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Select"
                local items = cfg.Values or {}
                local multi = cfg.Multi == true
                local flag = cfg.Flag
                local callback = cfg.Callback

                    AnchorPoint = Vector2.new(1, 0),

                    FontFace = Fonts.Body,

                    TextXAlignment = Enum.TextXAlignment.Right,

                    Parent = topBtn,
                })

                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -13, 0.5, 0),
                    Size = UDim2.fromOffset(14, 14),
                    BackgroundTransparency = 1,
                    Text = "v",
                    TextColor3 = Palette.TextMuted,
                    TextSize = 10,
                    FontFace = Fonts.Mono,
                    ZIndex = 19,
                    Parent = topBtn,
                })

                local list = make("Frame", {
                    Position = UDim2.fromOffset(10, 48),
                    Size = UDim2.new(1, -20, 0, 0),
                    AutomaticSize = Enum.AutomaticSize.Y,
                    BackgroundTransparency = 1,
                    ZIndex = 18,
                    Parent = row,
                }, {

                    make("UIPadding", { PaddingBottom = UDim.new(0, 10) }),

                local open = false
                local obj = {}

                local function summarize()
                    if multi then
                        local out = {}
                        for _, item in ipairs(items) do
                            if selected[item] then table.insert(out, tostring(item)) end
                        end
                        valLbl.Text = (#out > 0) and table.concat(out, ", ") or "None"
                    else
                        valLbl.Text = selected ~= nil and tostring(selected) or "None"
                    end
                    if flag then Velour.Flags[flag] = selected end
                end

                    open = state
                    local h = open and (58 + #items * 34) or 46
                    tween(row, 0.22, { Size = UDim2.new(1, 0, 0, h) })
                    tween(chev, 0.18, { Rotation = open and 180 or 0, TextColor3 = open and Palette.AccentBright or Palette.TextMuted })
                end

                    items = newItems or {}
                    for _, c in ipairs(list:GetChildren()) do
                        if c:IsA("TextButton") then c:Destroy() end

                        local isSel = multi and selected[item] or (selected == item)
                        local opt = make("TextButton", {
                            Size = UDim2.new(1, 0, 0, 30),
                            BackgroundColor3 = isSel and Palette.AccentSoft or Palette.Hairline,

                            if multi then
                                selected[item] = not selected[item]
                                local active = selected[item]
                                tween(opt, 0.14, {
                                    BackgroundColor3 = active and Palette.AccentSoft or Palette.Hairline,
                                    BackgroundTransparency = active and 0.28 or 0.93,
                                    TextColor3 = active and Palette.TextPrimary or Palette.TextSecondary,
                                })
                                summarize()
                                if callback then task.spawn(callback, selected) end
                            else
                                selected = item
                                summarize()
                                obj:Refresh(items)
                                setOpen(false)
                                if callback then task.spawn(callback, selected) end
                            end
                            silk:TriggerPulse(opt.AbsolutePosition + opt.AbsoluteSize * 0.5, 0.85)
                        end)
                    end
                    if open then setOpen(true) end
                end

                function obj:Set(val, silent)
                    selected = val
                    summarize()
                    obj:Refresh(items)
                    if not silent and callback then task.spawn(callback, selected) end
                end

                function obj:Get() return selected end

                topBtn.MouseButton1Click:Connect(function()
                    setOpen(not open)
                    silk:TriggerPulse(topBtn.AbsolutePosition + topBtn.AbsoluteSize * 0.5, 0.85)
                end)

                summarize()
                obj:Refresh(items)
                return obj
            end

--------------------------------------------------------------------------------
-- Config Export
--------------------------------------------------------------------------------

function Velour:ExportConfig()
    local out = {}
    for k, v in pairs(self.Flags) do
        local t = typeof(v)
        if t == "boolean" or t == "number" or t == "string" or t == "table" then
            out[k] = v
        elseif t == "EnumItem" then
            out[k] = v.Name
        end
    end
    return HttpService:JSONEncode(out)
end

return Velour
