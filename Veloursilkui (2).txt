local function getSafeParent()
    if type(gethui) == "function" then
        local ok, ui = pcall(gethui)
        if ok and ui then return ui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then return cg end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function make(className, props, children)
    local inst = Instance.new(className)
    if props then
        for k, v in pairs(props) do
            if k ~= "Parent" then
                inst[k] = v
            end
        end
    end
    if children then
        for _, c in ipairs(children) do
            c.Parent = inst
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function clamp(x, lo, hi)
    if x < lo then return lo end
    if x > hi then return hi end
    return x
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

local function tween(obj, duration, props, style, dir)
    local t = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

-- Tactile press-compression + localized crimson silk ripple inside any control
local function attachTactileFeedback(button, targetFrame, rippleParent, scaleAmount)
    local uiScale = targetFrame:FindFirstChildOfClass("UIScale") or make("UIScale", {
        Scale = 1,
        Parent = targetFrame,
    })

    local pressScale = scaleAmount or 0.975

    local function spawnRowRipple(screenPos)
        if not rippleParent then return end
        local absPos = rippleParent.AbsolutePosition
        local absSize = rippleParent.AbsoluteSize
        local rx = screenPos and (screenPos.X - absPos.X) or (absSize.X * 0.5)
        local ry = screenPos and (screenPos.Y - absPos.Y) or (absSize.Y * 0.5)
        local maxD = math.max(absSize.X, absSize.Y, 80) * 1.45

        local circle = make("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset(rx, ry),
            Size = UDim2.fromOffset(8, 8),
            BackgroundColor3 = Palette.RoseGlint,
            BackgroundTransparency = 0.72,
            BorderSizePixel = 0,
            ZIndex = (rippleParent.ZIndex or 17) + 1,
            Parent = rippleParent,
        }, {
            make("UICorner", { CornerRadius = UDim.new(1, 0) }),
        })

        local tw = tween(circle, 0.42, {
            Size = UDim2.fromOffset(maxD, maxD),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        tw.Completed:Once(function()
            if circle then circle:Destroy() end
        end)
    end

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(uiScale, 0.11, { Scale = pressScale }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            spawnRowRipple(input.Position)
        end
    end)

    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(uiScale, 0.22, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        end
    end)

    button.MouseLeave:Connect(function()
        tween(uiScale, 0.18, { Scale = 1 })
    end)
end

local Maid = {}
Maid.__index = Maid

function Maid.new()
    return setmetatable({ _items = {} }, Maid)
end

function Maid:Give(item)
    table.insert(self._items, item)
    return item
end

function Maid:Clean()
    for i = #self._items, 1, -1 do
        local item = self._items[i]
        self._items[i] = nil
        local t = typeof(item)
        if t == "RBXScriptConnection" then
            item:Disconnect()
        elseif t == "Instance" then
            item:Destroy()
        elseif t == "function" then
            pcall(item)
        end
    end
end

local Spring = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping)
    return setmetatable({
        Position = initial or 0,
        Target = initial or 0,
        Velocity = 0,
        Stiffness = stiffness or 250,
        Damping = damping or 23,
    }, Spring)
end

function Spring:Step(dt)
    local d = math.min(dt, 1 / 30)
    local force = (self.Target - self.Position) * self.Stiffness
    self.Velocity = self.Velocity + (force - self.Velocity * self.Damping) * d
    self.Position = self.Position + self.Velocity * d
    return self.Position
end

local LiquidSilk = {}
LiquidSilk.__index = LiquidSilk

local RIBBON_COUNT = 12
local TRAIL_COUNT  = 6
local MAX_PULSES   = 3

local function organicWave(x, y, t)
    local a = math.sin(x * 1.3 + t * 0.78) * math.cos(y * 1.1 - t * 0.56)
    local b = math.sin((x - y) * 2.1 + t * 1.12 + a * 0.8) * 0.5
    local c = math.cos((x * 1.8 + y * 2.3) - t * 1.48 + b * 0.6) * 0.25
    return (a + b + c) / 1.75
end

function LiquidSilk.new(hostFrame, maid, cornerRadius)
    local self = setmetatable({
        Host = hostFrame,
        Maid = maid,
        Time = 0,
        Active = true,
        PtrTarget = Vector2.new(0.5, 0.5),
        PtrCurrent = Vector2.new(0.5, 0.5),
        PtrVelocity = Vector2.new(0, 0),
        PtrEnergy = 0,
        PtrInside = false,
        PtrDown = false,
        Ribbons = {},
        Trails = {},
        Pulses = {},
        PulseCursor = 1,
    }, LiquidSilk)

    for i = 1, MAX_PULSES do
        self.Pulses[i] = { x = 0.5, y = 0.5, age = 10, strength = 0 }
    end

    -- Clipped silk container matching the exact strong rounded corners of the window
    local canvas = make("Frame", {
        Name = "LiquidSilkCanvas",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        ZIndex = 1,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
    })
    self.Canvas = canvas

    -- Full-bleed base gradient (guarantees zero black edge gaps at any angle)
    self.BaseGrad = make("UIGradient", {
        Rotation = 26,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.Base),
            ColorSequenceKeypoint.new(0.28, Palette.DeepSilk),
            ColorSequenceKeypoint.new(0.52, Palette.Oxblood),
            ColorSequenceKeypoint.new(0.76, Palette.DeepSilk),
            ColorSequenceKeypoint.new(1.00, Palette.Base),
        }),
        Parent = canvas,
    })

    -- 12 soft-feathered overlapping silk billows (zero sharp geometric edges)
    for i = 1, RIBBON_COUNT do
        local frac = (i - 1) / (RIBBON_COUNT - 1)
        local isCrest = (i % 2 == 0)
        local isHighlight = (i % 3 == 0)

        local ribbon = make("Frame", {
            Name = "SilkBillow_" .. i,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, frac),
            Size = UDim2.fromScale(2.15, 0.36 + (i % 3) * 0.09),
            BackgroundColor3 = isHighlight and Palette.Scarlet or (isCrest and Palette.Crimson or Palette.Oxblood),
            BackgroundTransparency = 0.72,
            BorderSizePixel = 0,
            Rotation = -18 + frac * 24,
            ZIndex = 2,
            Parent = canvas,
        }, {
            make("UICorner", { CornerRadius = UDim.new(1, 0) }),
        })

        local grad = make("UIGradient", {
            Rotation = 90,
            Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0.00, Palette.Base),
                ColorSequenceKeypoint.new(0.30, isCrest and Palette.Oxblood or Palette.DeepSilk),
                ColorSequenceKeypoint.new(0.50, isHighlight and Palette.RoseGlint or (isCrest and Palette.Scarlet or Palette.Crimson)),
                ColorSequenceKeypoint.new(0.70, isCrest and Palette.Oxblood or Palette.DeepSilk),
                ColorSequenceKeypoint.new(1.00, Palette.Base),
            }),
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 1.00),
                NumberSequenceKeypoint.new(0.22, 0.88),
                NumberSequenceKeypoint.new(0.50, isHighlight and 0.62 or 0.70),
                NumberSequenceKeypoint.new(0.78, 0.88),
                NumberSequenceKeypoint.new(1.00, 1.00),
            }),
            Parent = ribbon,
        })

        self.Ribbons[i] = {
            Frame = ribbon,
            Grad = grad,
            BaseY = -0.05 + frac * 1.10,
            BaseRot = -20 + frac * 28,
            Phase = i * 0.94,
            Speed = 0.55 + (i % 4) * 0.14,
            Highlight = isHighlight,
            Crest = isCrest,
        }
    end

    -- 6 elastic finger-trail silk sheen pools (creates buttery liquid wake when swiping)
    for i = 1, TRAIL_COUNT do
        local falloff = (i - 1) / (TRAIL_COUNT - 1)
        local sz = lerp(195, 90, falloff)
        local pool = make("Frame", {
            Name = "SilkWake_" .. i,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = UDim2.fromOffset(sz, sz * 0.72),
            BackgroundColor3 = (i <= 2) and Palette.RoseGlint or Palette.Scarlet,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ZIndex = 3,
            Parent = canvas,
        }, {
            make("UICorner", { CornerRadius = UDim.new(1, 0) }),
            make("UIGradient", {
                Rotation = 35 + i * 24,
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0.00, Palette.RoseGlint),
                    ColorSequenceKeypoint.new(0.45, Palette.Crimson),
                    ColorSequenceKeypoint.new(1.00, Palette.Base),
                }),
                Transparency = NumberSequence.new({
                    NumberSequenceKeypoint.new(0.00, 0.25),
                    NumberSequenceKeypoint.new(0.50, 0.72),
                    NumberSequenceKeypoint.new(1.00, 1.00),

            }),
        })

        self.Trails[i] = {
            Frame = pool,
            Pos = Vector2.new(0.5, 0.5),
            BaseSize = sz,
            FollowSpeed = lerp(18, 4.5, falloff),
            MaxOpacity = lerp(0.68, 0.90, falloff),

    self.RippleVisuals = {}
    for i = 1, MAX_PULSES do
        local ring = make("Frame", {
            Name = "CreaseWave_" .. i,

            make("UICorner", { CornerRadius = UDim.new(1, 0) }),

            Color = Palette.RoseGlint,
            Thickness = 2.6,
            Transparency = 1,
            Parent = ring,

        local grad = make("UIGradient", {
            Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0.0, Palette.RoseGlint),
                ColorSequenceKeypoint.new(0.5, Palette.Crimson),

        })

    -- Soft dark edge vignette for depth & crisp text contrast
    make("Frame", {
        Name = "SilkVignette",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BackgroundTransparency = 0.28,
        BorderSizePixel = 0,
        ZIndex = 5,
        Parent = canvas,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
        make("UIGradient", {
            Rotation = 90,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 0.28),
                NumberSequenceKeypoint.new(0.45, 0.62),
                NumberSequenceKeypoint.new(1.00, 0.20),
            }),
        }),
    })

    self:_bindInput()
    self:_startLoop()
    return self
end

function LiquidSilk:_toNorm(screenPos)
    local absPos = self.Host.AbsolutePosition
    local absSize = self.Host.AbsoluteSize
    if absSize.X <= 1 or absSize.Y <= 1 then
        return 0.5, 0.5, false
    end
    local nx = (screenPos.X - absPos.X) / absSize.X
    local ny = (screenPos.Y - absPos.Y) / absSize.Y
    local inside = nx >= 0 and nx <= 1 and ny >= 0 and ny <= 1
    return clamp(nx, 0, 1), clamp(ny, 0, 1), inside
end

function LiquidSilk:TriggerPulse(screenPos, strength)
    local nx, ny = 0.5, 0.5
    if screenPos then
        nx, ny = self:_toNorm(screenPos)
    end
    local slot = self.Pulses[self.PulseCursor]
    slot.x = nx
    slot.y = ny
    slot.age = 0
    slot.strength = strength or 1.0
    self.PulseCursor = (self.PulseCursor % MAX_PULSES) + 1
end

function LiquidSilk:_bindInput()
    self.Maid:Give(UserInputService.InputChanged:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local nx, ny, inside = self:_toNorm(input.Position)
            self.PtrInside = inside
            if inside then
                self.PtrTarget = Vector2.new(nx, ny)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputBegan:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local nx, ny, inside = self:_toNorm(input.Position)
            if inside then
                self.PtrInside = true
                self.PtrDown = true
                self.PtrTarget = Vector2.new(nx, ny)
                self:TriggerPulse(input.Position, 1.0)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            self.PtrDown = false
        end
    end))
end

function LiquidSilk:_startLoop()
    self.Maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not self.Active or not self.Host.Visible then return end
        local step = math.min(dt, 0.05)
        self.Time = self.Time + step
        local t = self.Time

        local prev = self.PtrCurrent
        self.PtrCurrent = self.PtrCurrent:Lerp(self.PtrTarget, 1 - math.exp(-step * 14))
        local vel = (self.PtrCurrent - prev) / math.max(step, 1 / 120)
        self.PtrVelocity = self.PtrVelocity:Lerp(vel, 1 - math.exp(-step * 9))

        local targetE = self.PtrDown and 1.25 or (self.PtrInside and 0.78 or 0)
        self.PtrEnergy = lerp(self.PtrEnergy, targetE, 1 - math.exp(-step * 7))

        for i = 1, MAX_PULSES do
            local p = self.Pulses[i]
            if p.strength > 0 then
                p.age = p.age + step
                if p.age > 1.35 then p.strength = 0 end
            end
        end

        local px, py = self.PtrCurrent.X, self.PtrCurrent.Y
        local vx, vy = self.PtrVelocity.X, self.PtrVelocity.Y
        local swipeSpeed = math.min(self.PtrVelocity.Magnitude, 3.0)
        local energy = self.PtrEnergy

        -- 1. Flowing base silk gradient
        local k1 = clamp(0.24 + 0.12 * math.sin(t * 0.52) + (px - 0.5) * 0.09 * energy, 0.10, 0.36)
        local k2 = clamp(0.52 + 0.14 * math.sin(t * 0.44 + 1.3) + (py - 0.5) * 0.09 * energy, k1 + 0.10, 0.70)
        local k3 = clamp(0.80 + 0.08 * math.cos(t * 0.38 + 0.6), k2 + 0.10, 0.92)
        self.BaseGrad.Rotation = 22 + math.sin(t * 0.34) * 16 + (px - 0.5) * 20 * energy
        self.BaseGrad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.Base),
            ColorSequenceKeypoint.new(k1, Palette.DeepSilk),
            ColorSequenceKeypoint.new(k2, Palette.Crimson),
            ColorSequenceKeypoint.new(k3, Palette.Oxblood),
            ColorSequenceKeypoint.new(1.00, Palette.Base),
        })

        -- 2. Wind-blown silk ribbons + silky swipe push + Gaussian ripple perturbation
        for i = 1, RIBBON_COUNT do
            local rb = self.Ribbons[i]
            local w = organicWave(rb.Phase * 0.7, rb.BaseY * 2.4, t * rb.Speed)
            local windY = w * 0.068
            local windX = math.sin(t * 0.48 + rb.Phase) * 0.10
            local windRot = w * 8.5

            local dy = rb.BaseY - py
            local influence = math.exp(-(dy * dy) / 0.048) * energy
            local swipePush = (-dy * 0.13 + clamp(vy * 0.04, -0.08, 0.08)) * influence
            local swipeTilt = ((px - 0.5) * 16 + clamp(vx * 8, -14, 14)) * influence

            local ripplePush = 0
            local rippleGlow = 0
            for j = 1, MAX_PULSES do
                local pulse = self.Pulses[j]
                if pulse.strength > 0 then
                    local pd = math.abs(rb.BaseY - pulse.y)
                    local front = (pd - pulse.age * 0.92) / 0.11
                    local env = math.exp(-(front * front)) * pulse.strength * math.exp(-pulse.age * 1.75)
                    local sign = (rb.BaseY >= pulse.y) and 1 or -1
                    ripplePush = ripplePush + sign * env * math.cos(front * 2.4) * 0.075
                    rippleGlow = rippleGlow + env
                end
            end

            rb.Frame.Position = UDim2.fromScale(
                0.5 + windX,
                clamp(rb.BaseY + windY + swipePush + ripplePush, -0.18, 1.18)
            )
            rb.Frame.Rotation = rb.BaseRot + windRot + swipeTilt + ripplePush * 110
            rb.Grad.Offset = Vector2.new(
                math.sin(t * 0.52 + rb.Phase) * 0.18,
                math.cos(t * 0.41 + rb.Phase) * 0.14
            )

            local baseTrans = rb.Highlight and 0.66 or (rb.Crest and 0.72 or 0.76)
            local boost = influence * (0.16 + swipeSpeed * 0.06) + rippleGlow * 0.25
            rb.Frame.BackgroundTransparency = clamp(baseTrans - boost, 0.32, 0.92)
        end

        -- 3. Buttery liquid finger-trail sheen pools
        local swipeAngle = math.deg(math.atan2(vy, vx))
        for i = 1, TRAIL_COUNT do
            local tr = self.Trails[i]
            tr.Pos = tr.Pos:Lerp(self.PtrCurrent, 1 - math.exp(-step * tr.FollowSpeed))
            local stretch = 1 + math.min(swipeSpeed * 0.35, 0.85)
            tr.Frame.Position = UDim2.fromScale(tr.Pos.X, tr.Pos.Y)
            tr.Frame.Size = UDim2.fromOffset(tr.BaseSize * stretch, tr.BaseSize * (0.72 / math.sqrt(stretch)))
            if swipeSpeed > 0.08 then
                tr.Frame.Rotation = lerp(tr.Frame.Rotation, swipeAngle, 1 - math.exp(-step * 10))
            end
            tr.Frame.BackgroundTransparency = lerp(1.0, tr.MaxOpacity, clamp(energy, 0, 1))
        end

        -- 4. Expanding crease wave rings
        local absSize = self.Host.AbsoluteSize
        local maxDim = math.max(absSize.X, absSize.Y, 240)
        for i = 1, MAX_PULSES do
            local pulse = self.Pulses[i]
            local vis = self.RippleVisuals[i]
            if pulse.strength > 0 then
                local prog = clamp(pulse.age / 1.1, 0, 1)
                local eased = 1 - math.pow(1 - prog, 3)
                local rad = eased * maxDim * 0.96
                local fade = math.exp(-pulse.age * 2.5) * math.min(pulse.age / 0.06, 1)
                vis.Ring.Visible = true
                vis.Ring.Position = UDim2.fromScale(pulse.x, pulse.y)
                vis.Ring.Size = UDim2.fromOffset(rad * 1.24, rad * 0.90)
                vis.Ring.Rotation = 14 + pulse.age * 20
                vis.Stroke.Thickness = lerp(2.8, 0.7, eased)
                vis.Stroke.Transparency = clamp(1 - fade * 0.72, 0.22, 1)
                vis.Grad.Rotation = (t * 85 + i * 110) % 360
            else
                vis.Ring.Visible = false
            end
        end
    end))
end

local function attachOceanWaveBorder(targetFrame, maid, thickness)
    local stroke = make("UIStroke", {
        Color = Color3.new(1, 1, 1),
        Thickness = thickness or 1.6,
        Transparency = 0.14,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = targetFrame,
    })

    local grad = make("UIGradient", {
        Rotation = 0,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.OceanAbyss),
            ColorSequenceKeypoint.new(0.28, Palette.OceanWave),
            ColorSequenceKeypoint.new(0.50, Palette.OceanCrest),
            ColorSequenceKeypoint.new(0.62, Palette.OceanFoam),
            ColorSequenceKeypoint.new(0.78, Palette.OceanWave),
            ColorSequenceKeypoint.new(1.00, Palette.OceanAbyss),
        }),
        Parent = stroke,
    })

    local elapsed = 0
    maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not targetFrame.Visible then return end
        elapsed = elapsed + math.min(dt, 0.05)
        -- Rolling ocean wave sweep around the pill perimeter
        local wave1 = 0.25 + 0.12 * math.sin(elapsed * 1.6)
        local wave2 = 0.50 + 0.14 * math.sin(elapsed * 1.3 + 1.1)
        local crest = clamp(wave2 + 0.11, wave2 + 0.05, 0.80)
        local tail  = clamp(crest + 0.12, crest + 0.05, 0.92)

        grad.Rotation = (elapsed * 42 + math.sin(elapsed * 1.1) * 24) % 360
        grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Palette.OceanAbyss),
            ColorSequenceKeypoint.new(clamp(wave1, 0.10, 0.34), Palette.OceanWave),
            ColorSequenceKeypoint.new(clamp(wave2, 0.38, 0.62), Palette.OceanCrest),
            ColorSequenceKeypoint.new(crest, Palette.OceanFoam),
            ColorSequenceKeypoint.new(tail, Palette.OceanWave),
            ColorSequenceKeypoint.new(1.00, Palette.OceanAbyss),
        })
    end))

    return stroke, grad
end

local function attachDrag(handle, target, onTap)
    local dragging = false
    local dragStart, startPos
    local moved = false
    local conns = {}

    table.insert(conns, handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))

    table.insert(conns, UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                moved = true
            end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end
    end))

    table.insert(conns, UserInputService.InputEnded:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            if not moved and onTap then
                onTap(input.Position)
            end
        end
    end))

    return function()
        for _, c in ipairs(conns) do
            c:Disconnect()
        end
    end
end

                    task.delay(0.10, function()
                        if knob and knob.Parent then
                            tween(knob, 0.18, { Size = UDim2.fromOffset(14, 14) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        end
                    end)

                    if value then
                        tween(track, 0.22, { BackgroundColor3 = Palette.Accent, BackgroundTransparency = 0.06 })
                        tween(knob, 0.24, {
                            Position = UDim2.new(1, -17, 0.5, 0),
                            BackgroundColor3 = Palette.TextPrimary,
                        }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    else
                        tween(track, 0.22, { BackgroundColor3 = Palette.Hairline, BackgroundTransparency = 0.86 })
                        tween(knob, 0.22, {
                            Position = UDim2.new(0, 3, 0.5, 0),
                            BackgroundColor3 = Palette.TextSecondary,
                        })
                    end

                    if not silent and callback then
                        task.spawn(callback, value)
                    end
                end

                function obj:Get() return value end

                hit.MouseButton1Click:Connect(function()
                    obj:Set(not value)
                    silk:TriggerPulse(track.AbsolutePosition + track.AbsoluteSize * 0.5, 0.75)
                end)

                hit.MouseEnter:Connect(function() tween(row, 0.14, { BackgroundTransparency = 0.50 }) end)
                hit.MouseLeave:Connect(function() tween(row, 0.14, { BackgroundTransparency = 1 }) end)

                obj:Set(value, true)
                return obj
            end

            function section:AddSlider(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Slider"
                local minV = cfg.Min or 0
                local maxV = cfg.Max or 100
                local step = cfg.Increment or 1
                local suffix = cfg.Suffix or ""
                local flag = cfg.Flag
                local value = clamp(cfg.Default or minV, minV, maxV)
                local callback = cfg.Callback

                local row = addRow(56)

                make("TextLabel", {
                    Position = UDim2.new(0, 14, 0, 9),
                    Size = UDim2.new(1, -86, 0, 16),
                    BackgroundTransparency = 1,
                    Text = label,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 12,
                    FontFace = Fonts.Body,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = row,
                })

                local readout = make("TextLabel", {
                    AnchorPoint = Vector2.new(1, 0),
                    Position = UDim2.new(1, -14, 0, 9),
                    Size = UDim2.fromOffset(70, 16),
                    BackgroundTransparency = 1,
                    Text = tostring(value) .. suffix,
                    TextColor3 = Palette.TextSecondary,
                    TextSize = 11,
                    FontFace = Fonts.Mono,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    ZIndex = 18,
                    Parent = row,
                })
                local readoutScale = make("UIScale", { Scale = 1, Parent = readout })

                local hit = make("TextButton", {
                    Position = UDim2.new(0, 12, 0, 26),
                    Size = UDim2.new(1, -24, 0, 26),
                    BackgroundTransparency = 1,
                    Text = "",
                    AutoButtonColor = false,
                    ZIndex = 19,
                    Parent = row,
                })

                local rail = make("Frame", {
                    AnchorPoint = Vector2.new(0, 0.5),
                    Position = UDim2.fromScale(0, 0.5),
                    Size = UDim2.new(1, 0, 0, 4),
                    BackgroundColor3 = Palette.Hairline,
                    BackgroundTransparency = 0.86,
                    BorderSizePixel = 0,
                    ZIndex = 18,
                    Parent = hit,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local fill = make("Frame", {
                    Size = UDim2.fromScale(0, 1),
                    BackgroundColor3 = Palette.Accent,
                    BorderSizePixel = 0,
                    ZIndex = 19,
                    Parent = rail,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                    make("UIGradient", {
                        Color = ColorSequence.new({
                            ColorSequenceKeypoint.new(0, Palette.Oxblood),
                            ColorSequenceKeypoint.new(0.65, Palette.Crimson),
                            ColorSequenceKeypoint.new(1, Palette.RoseGlint),
                        }),
                    }),
                })

                local thumbHalo = make("Frame", {
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromScale(1, 0.5),
                    Size = UDim2.fromOffset(12, 12),
                    BackgroundColor3 = Palette.RoseGlint,
                    BackgroundTransparency = 1,
                    BorderSizePixel = 0,
                    ZIndex = 19,
                    Parent = fill,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local thumb = make("Frame", {
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromScale(1, 0.5),
                    Size = UDim2.fromOffset(12, 12),
                    BackgroundColor3 = Palette.TextPrimary,
                    BorderSizePixel = 0,
                    ZIndex = 20,
                    Parent = fill,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local obj = {}
                local dragging = false
                local activeInput = nil

                local function fmt(n)
                    if step < 1 then
                        local d = math.max(1, math.ceil(-math.log10(step)))
                        return string.format("%." .. d .. "f", n)
                    end
                    return tostring(math.floor(n + 0.5))
                end

                function obj:Set(v, silent)
                    local snapped = math.floor(((v - minV) / step) + 0.5) * step + minV
                    value = clamp(snapped, minV, maxV)
                    if flag then Velour.Flags[flag] = value end
                    local pct = (maxV == minV) and 0 or (value - minV) / (maxV - minV)
                    readout.Text = fmt(value) .. suffix
                    tween(fill, 0.05, { Size = UDim2.fromScale(pct, 1) })
                    if not silent and callback then
                        task.spawn(callback, value)
                    end
                end

                function obj:Get() return value end

                local function updateX(x)
                    local rel = (x - rail.AbsolutePosition.X) / math.max(rail.AbsoluteSize.X, 1)
                    obj:Set(minV + clamp(rel, 0, 1) * (maxV - minV))
                end

                local function endDrag()
                    if not dragging then return end
                    dragging = false
                    activeInput = nil
                    page.ScrollingEnabled = true
                    tween(thumb, 0.16, { Size = UDim2.fromOffset(12, 12) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    tween(thumbHalo, 0.18, { Size = UDim2.fromOffset(12, 12), BackgroundTransparency = 1 })
                    tween(readout, 0.16, { TextColor3 = Palette.TextSecondary })
                    tween(readoutScale, 0.16, { Scale = 1 })
                end

                hit.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = true
                        activeInput = input
                        page.ScrollingEnabled = false
                        tween(thumb, 0.14, { Size = UDim2.fromOffset(15, 15) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        tween(thumbHalo, 0.18, { Size = UDim2.fromOffset(26, 26), BackgroundTransparency = 0.72 })
                        tween(readout, 0.14, { TextColor3 = Palette.AccentBright })
                        tween(readoutScale, 0.14, { Scale = 1.08 })
                        updateX(input.Position.X)
                        silk:TriggerPulse(input.Position, 0.65)
                    end
                end)

                maid:Give(UserInputService.InputChanged:Connect(function(input)
                    if not dragging then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement
                        or (input.UserInputType == Enum.UserInputType.Touch and (activeInput == nil or input == activeInput)) then
                        updateX(input.Position.X)
                    end
                end))

                maid:Give(UserInputService.InputEnded:Connect(function(input)
                    if not dragging then return end
                    if input == activeInput
                        or input.UserInputType == Enum.UserInputType.MouseButton1
                        or input.UserInputType == Enum.UserInputType.Touch then
                        endDrag()
                    end
                end))

                obj:Set(value, true)
                return obj
            end

            function section:AddDropdown(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Select"
                local items = cfg.Values or {}
                local multi = cfg.Multi == true
                local flag = cfg.Flag
                local callback = cfg.Callback

                local selected = multi and {} or cfg.Default

                    FontFace = Fonts.Reg,
                    TextTruncate = Enum.TextTruncate.AtEnd,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    ZIndex = 19,
                    Parent = topBtn,
                })

                local chev = make("TextLabel", {
                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -12, 0.5, 0),
                    Size = UDim2.fromOffset(14, 14),
                    BackgroundTransparency = 1,
                    Text = "v",
                    TextColor3 = Palette.TextMuted,
                    TextSize = 9,
                    FontFace = Fonts.Mono,
                    ZIndex = 19,
                    Parent = topBtn,
                })

                local list = make("Frame", {
                    Position = UDim2.fromOffset(10, 44),
                    Size = UDim2.new(1, -20, 0, 0),
                    AutomaticSize = Enum.AutomaticSize.Y,
                    BackgroundTransparency = 1,
                    ZIndex = 18,
                    Parent = row,
                }, {

                    make("UIPadding", { PaddingBottom = UDim.new(0, 10) }),
                })

                local open = false
                local obj = {}

                local function summarize()
                    if multi then
                        local out = {}
                        for _, item in ipairs(items) do
                            if selected[item] then table.insert(out, tostring(item)) end
                        end
                        valLbl.Text = (#out > 0) and table.concat(out, ", ") or "None"
                    else
                        valLbl.Text = selected ~= nil and tostring(selected) or "None"
                    end
                    if flag then Velour.Flags[flag] = selected end
                end

                    open = state
                    local h = open and (54 + #items * 32) or 42
                    tween(row, 0.24, { Size = UDim2.new(1, 0, 0, h) })
                    tween(chev, 0.20, { Rotation = open and 180 or 0, TextColor3 = open and Palette.AccentBright or Palette.TextMuted })
                end

                function obj:Refresh(newItems)
                    items = newItems or {}
                    for _, c in ipairs(list:GetChildren()) do
                        if c:IsA("TextButton") then c:Destroy() end
                    end
                    for _, item in ipairs(items) do
                        local isSel = multi and selected[item] or (selected == item)
                        local opt = make("TextButton", {
                            Size = UDim2.new(1, 0, 0, 28),

                        opt.MouseButton1Click:Connect(function()
                            if multi then
                                selected[item] = not selected[item]
                                local active = selected[item]
                                tween(opt, 0.14, {
                                    BackgroundColor3 = active and Palette.AccentSoft or Palette.Hairline,
                                    BackgroundTransparency = active and 0.30 or 0.94,
                                    TextColor3 = active and Palette.TextPrimary or Palette.TextSecondary,
                                })
                                summarize()
                                if callback then task.spawn(callback, selected) end
                            else
                                selected = item
                                summarize()
                                obj:Refresh(items)
                                setOpen(false)
                                if callback then task.spawn(callback, selected) end
                            end
                            silk:TriggerPulse(opt.AbsolutePosition + opt.AbsoluteSize * 0.5, 0.65)
                        end)
                    end
                    if open then setOpen(true) end
                end

                function obj:Set(val, silent)
                    selected = val
                    summarize()
                    obj:Refresh(items)
                    if not silent and callback then task.spawn(callback, selected) end
                end

                function obj:Get() return selected end

                topBtn.MouseButton1Click:Connect(function()
                    setOpen(not open)
                    silk:TriggerPulse(topBtn.AbsolutePosition + topBtn.AbsoluteSize * 0.5, 0.65)
                end)

                summarize()
                obj:Refresh(items)
                return obj
            end

--------------------------------------------------------------------------------
-- Config Export
--------------------------------------------------------------------------------

function Velour:ExportConfig()
    local out = {}
    for k, v in pairs(self.Flags) do
        local t = typeof(v)
        if t == "boolean" or t == "number" or t == "string" or t == "table" then
            out[k] = v
        elseif t == "EnumItem" then
            out[k] = v.Name
        end
    end
    return HttpService:JSONEncode(out)
end

return Velour
