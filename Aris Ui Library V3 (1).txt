-- Hàm chuyển đổi thông minh hỗ trợ cả Decal ID và Image ID (Mặc định: 5205043242)
function Lib.ResolveDecalImage(idInput)
    if not idInput or idInput == "" then
        return "rbxthumb://type=Asset&id=" .. Lib.DefaultDecalId .. "&w=420&h=420", "rbxassetid://" .. Lib.DefaultDecalId
    end
    local s = tostring(idInput)
    local digits = string.match(s, "^%s*(%d+)%s*$")
        or string.match(s, "rbxassetid://(%d+)")
        or string.match(s, "id=(%d+)")
    if digits then
        return "rbxthumb://type=Asset&id=" .. digits .. "&w=420&h=420", "rbxassetid://" .. digits
    end
    return s, s
end

-- Hàm dọn dẹp triệt để mọi Menu cũ trên màn hình (chống lỗi Double Menu khi chèn script)
function Lib.DestroyAllExistingMenus(customGuiName)
    for _, w in ipairs(Lib.Windows) do
        if w.Connections then
            for _, c in ipairs(w.Connections) do
                pcall(function() c:Disconnect() end)
            end
            table.clear(w.Connections)
        end
        if w.UI and w.UI.ScreenGui then
            pcall(function() w.UI.ScreenGui:Destroy() end)
        end
        if w.UI and w.UI.MenuBlur then
            pcall(function() w.UI.MenuBlur:Destroy() end)
        end
    end
    table.clear(Lib.Windows)

    local containers = { Lib.TargetGui }
    pcall(function()
        local cg = game:GetService("CoreGui")
        if cg and not table.find(containers, cg) then table.insert(containers, cg) end
    end)
    pcall(function()
        local pg = Lib.LocalPlayer and Lib.LocalPlayer:FindFirstChild("PlayerGui")
        if pg and not table.find(containers, pg) then table.insert(containers, pg) end
    end)

    local knownNames = {
        ["Aris_UILibrary_V3"] = true,
        ["Aris_Hub_V3_UI"] = true,
    }
    if customGuiName then knownNames[tostring(customGuiName)] = true end

    for _, parentGui in ipairs(containers) do
        pcall(function()
            for _, child in ipairs(parentGui:GetChildren()) do
                if child:IsA("ScreenGui") then
                    local isAris = knownNames[child.Name]
                        or child:GetAttribute("IsArisUILibrary") == true
                        or child:FindFirstChild("__ArisUILibMarker") ~= nil
                        or (child:FindFirstChild("mainmage") ~= nil and child:FindFirstChild("icon") ~= nil)
                    if isAris then
                        child:Destroy()
                    end
                end
            end
        end)
    end

    pcall(function()
        for _, eff in ipairs(Lib.Lighting:GetChildren()) do
            if eff.Name == "Aris_UILib_MenuBlur" then
                eff:Destroy()
            end
        end
    end)
end

-- PHẦN 2: HÀM TIỆN ÍCH LƯU/NẠP CONFIG & CHUYỂN ĐỔI MÀU SẮC (COLOR3 <-> HEX)
-- =================================================================================
function Lib.ColorToHex(col)
    if typeof(col) ~= "Color3" then return "#FFFFFF" end
    local r = math.clamp(math.floor(col.R * 255 + 0.5), 0, 255)
    local g = math.clamp(math.floor(col.G * 255 + 0.5), 0, 255)
    local b = math.clamp(math.floor(col.B * 255 + 0.5), 0, 255)
    return string.format("#%02X%02X%02X", r, g, b)
end

function Lib.HexToColor(hexStr)
    if type(hexStr) ~= "string" then return nil end
    local clean = string.gsub(hexStr, "^#", "")
    clean = string.gsub(clean, "%s+", "")
    if #clean == 6 then
        local r = tonumber(string.sub(clean, 1, 2), 16)
        local g = tonumber(string.sub(clean, 3, 4), 16)
        local b = tonumber(string.sub(clean, 5, 6), 16)
        if r and g and b then
            return Color3.fromRGB(r, g, b)
        end
    end
    return nil
end

function Lib.SerializeValue(v)
    if typeof(v) == "Color3" then
        return { __type = "Color3", R = math.floor(v.R * 255 + 0.5), G = math.floor(v.G * 255 + 0.5), B = math.floor(v.B * 255 + 0.5) }
    elseif typeof(v) == "EnumItem" then
        return { __type = "KeyCode", Name = v.Name }
    elseif type(v) == "table" then
        local out = {}
        for k, sub in pairs(v) do
            out[k] = Lib.SerializeValue(sub)
        end
        return out
    elseif type(v) == "number" or type(v) == "boolean" or type(v) == "string" then
        return v
    end
    return nil
end

function Lib.DeserializeValue(v)
    if type(v) == "table" then
        if v.__type == "Color3" then
            return Color3.fromRGB(v.R or 255, v.G or 255, v.B or 255)
        elseif v.__type == "KeyCode" and v.Name and Enum.KeyCode[v.Name] then
            return Enum.KeyCode[v.Name]
        else
            local out = {}
            for k, sub in pairs(v) do
                out[k] = Lib.DeserializeValue(sub)
            end
            return out
        end
    end
    return v
end

function Lib.EnsureFolders(rootFolder)
    pcall(function()
        if type(isfolder) == "function" and type(makefolder) == "function" then
            if not isfolder(rootFolder) then makefolder(rootFolder) end
            if not isfolder(rootFolder .. "/Configs") then makefolder(rootFolder .. "/Configs") end
        end
    end)
end

function Lib.FormatHMS(sec)
    sec = math.max(0, math.floor(sec or 0))
    return string.format("%02d:%02d:%02d", math.floor(sec / 3600), math.floor((sec % 3600) / 60), sec % 60)
end

    function Win:Notify(setting)
        setting = setting or {}
        local title = setting.Title or ""
        local desc = setting.Description or setting.Desc or setting.Content or ""
        local dur = tonumber(setting.Duration) or 2.6

        Win.NotifToken = Win.NotifToken + 1
        local myToken = Win.NotifToken

        local prefix = (title ~= "") and ("[" .. tostring(title) .. "] ") or ""
        U.NotifTitle.Text = prefix .. tostring(desc)
        U.Notif.Visible = true
        U.Notif.Position = UDim2.new(0.5, 0, -0.12, 0)

        Lib.TweenService:Create(U.Notif, TweenInfo.new(0.32, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Position = UDim2.new(0.5, 0, 0.075, 0)
        }):Play()

        task.delay(dur, function()
            if Win.NotifToken ~= myToken then return end
            Lib.TweenService:Create(U.Notif, TweenInfo.new(0.24, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Position = UDim2.new(0.5, 0, -0.12, 0)
            }):Play()
            task.delay(0.25, function()
                if Win.NotifToken == myToken and U.Notif then
                    U.Notif.Visible = false
                end
            end)
        end)
    end

    -- HÀM ANIMATION TRƯỢT CÁC NÚT KHI CHUYỂN TAB HOẶC MỞ/ĐÓNG TAB
    -- =================================================================================
    function Win:PlayButtonsSlideIn(scrollOrFrame)
        if not scrollOrFrame then return end
        Win.SlideAnimToken = Win.SlideAnimToken + 1
        local token = Win.SlideAnimToken
        local idx = 0
        local baseDur = math.clamp(S.AnimationSpeed or 0.24, 0.12, 0.55)

        for _, slot in ipairs(scrollOrFrame:GetChildren()) do
            if slot:IsA("GuiObject") and slot.Visible then
                local animTarget = slot:FindFirstChild("AnimBody") or slot:FindFirstChildOfClass("Frame") or slot:FindFirstChildOfClass("TextLabel")
                if animTarget and animTarget:IsA("GuiObject") then
                    idx = idx + 1
                    local delayTime = math.min((idx - 1) * 0.022, 0.26)
                    local targetY = animTarget:GetAttribute("RestYScale") or 0
                    local startOffset = 38 + math.min(idx * 4, 28)
                    animTarget.Position = UDim2.new(0, startOffset, targetY, 0)

                    task.delay(delayTime, function()
                        if Win.SlideAnimToken ~= token or not animTarget.Parent then return end
                        Win:Tween(animTarget, {
                            Position = UDim2.new(0, 0, targetY, 0)
                        }, baseDur + 0.06, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    end)
                end
            end
        end
    end

    function Win:PlayNavButtonsSlide(isOpening)
        local baseDur = math.clamp(S.AnimationSpeed or 0.24, 0.14, 0.50)
        for i, key in ipairs(Win.LeftTabOrderList) do
            local info = Win.LeftTabs[key]
            if info and info.Inner and info.Inner.Parent then
                if isOpening then
                    info.Inner.Position = UDim2.new(-0.35, 0, 0, 0)
                    task.delay(math.min((i - 1) * 0.025, 0.22), function()
                        if info.Inner and info.Inner.Parent then
                            Win:Tween(info.Inner, { Position = UDim2.new(0, 0, 0, 0) }, baseDur + 0.06, Enum.EasingStyle.Back)
                        end
                    end)
                else
                    Win:Tween(info.Inner, { Position = UDim2.new(-0.25, 0, 0, 0) }, 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
                end
            end
        end

        for i, key in ipairs(Win.BottomTabOrderList) do
            local bInfo = Win.BottomTabs[key]
            if bInfo and bInfo.Inner and bInfo.Inner.Parent then
                if isOpening then
                    bInfo.Inner.Position = UDim2.new(0, 0, 0.65, 0)
                    task.delay(math.min((i - 1) * 0.025, 0.22), function()
                        if bInfo.Inner and bInfo.Inner.Parent then
                            Win:Tween(bInfo.Inner, { Position = UDim2.new(0, 0, 0, 0) }, baseDur + 0.06, Enum.EasingStyle.Back)
                        end
                    end)
                else
                    Win:Tween(bInfo.Inner, { Position = UDim2.new(0, 0, 0.55, 0) }, 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
                end
            end
        end
    end

    -- Kéo thả Cửa Sổ & Nút Kéo Phóng To Menu
    function Win:MakeDraggable(frame, handle, saveMenuPos)
        local dragging = false
        local dragStart, startPos
        local targetHandle = handle or frame

        Win:BindConnection(targetHandle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = frame.Position
            end
        end))

        Win:BindConnection(Lib.UserInputService.InputChanged:Connect(function(input)
            if not dragging then return end
            if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
            local delta = input.Position - dragStart
            local newPos = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
            frame.Position = newPos
            if saveMenuPos then
                Win.SavedMenuPos = newPos
            end
        end))

        Win:BindConnection(Lib.UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end))
    end

    Win:MakeDraggable(U.MainImage, U.MainBar, true)
    Win:MakeDraggable(U.FloatingIconFrame, U.FloatingIconBtn, false)

    -- Tìm kiếm realtime trên GridView & DetailView
    U.GridSearchBox:GetPropertyChangedSignal("Text"):Connect(function()
        local q = string.lower(U.GridSearchBox.Text or "")
        for _, slot in ipairs(U.GridScroll:GetChildren()) do
            if slot:IsA("Frame") then
                local titleObj = slot:FindFirstChild("title", true)
                local txt = titleObj and string.lower(titleObj.Text) or ""
                slot.Visible = (q == "" or string.find(txt, q, 1, true) ~= nil)
            end
        end
    end)

    U.DetailSearchBox:GetPropertyChangedSignal("Text"):Connect(function()
        local q = string.lower(U.DetailSearchBox.Text or "")
        for _, slot in ipairs(U.DetailScroll:GetChildren()) do
            if slot:IsA("GuiObject") then
                local match = (q == "")
                if not match then
                    for _, desc in ipairs(slot:GetDescendants()) do
                        if (desc:IsA("TextLabel") or desc:IsA("TextButton")) and string.find(string.lower(desc.Text), q, 1, true) then
                            match = true
                            break
                        end
                    end
                end
                slot.Visible = match
            end
        end
    end)

    -- Chuyển View kèm hiệu ứng trượt toàn bộ nút bên trong
    function Win:SwitchView(targetFrame)
        for _, v in ipairs({U.HomeView, U.GridView, U.DetailView, U.FeedbackView}) do
            if v then v.Visible = (v == targetFrame) end
        end
        if targetFrame then
            targetFrame.Position = UDim2.new(0.045, 0, 0, 0)
            Win:Tween(targetFrame, { Position = UDim2.new(0, 0, 0, 0) }, S.AnimationSpeed, Enum.EasingStyle.Quart)
            if targetFrame == U.DetailView then
                Win:PlayButtonsSlideIn(U.DetailScroll)
            elseif targetFrame == U.GridView then
                Win:PlayButtonsSlideIn(U.GridScroll)
            elseif targetFrame == U.HomeView then
                Win:PlayButtonsSlideIn(U.HomeView)
            end
        end
    end

    function Win:HighlightLeftTab(activeKey)
        for key, info in pairs(Win.LeftTabs) do
            local isAct = (key == activeKey)
            Win:Tween(info.Inner, {
                BackgroundTransparency = isAct and 0.12 or 0.78,
                Position = isAct and UDim2.new(0, 6, 0, 0) or UDim2.new(0, 0, 0, 0),
            }, S.AnimationSpeed, Enum.EasingStyle.Back)
            Win:Tween(info.Button, {
                Position = isAct and UDim2.new(0, 18, 0.5, 0) or UDim2.new(0, 12, 0.5, 0)
            }, S.AnimationSpeed, Enum.EasingStyle.Back)
            if info.Indicator then
                Win:Tween(info.Indicator, {
                    Size = isAct and UDim2.new(0, 4, 0.62, 0) or UDim2.new(0, 0, 0.2, 0),
                    BackgroundTransparency = isAct and 0 or 1,
                }, S.AnimationSpeed, Enum.EasingStyle.Back)
            end
        end
        for _, bInfo in pairs(Win.BottomTabs) do
            Win:Tween(bInfo.Inner, {
                BackgroundColor3 = Color3.fromRGB(49, 52, 60),
                BackgroundTransparency = 0.28,
                Position = UDim2.new(0, 0, 0, 0),
            }, S.AnimationSpeed)
        end
    end

            plusBtn.TextScaled = true
            plusBtn.Text = "+"
            Instance.new("UICorner", plusBtn).CornerRadius = UDim.new(0.28, 0)
            Instance.new("UITextSizeConstraint", plusBtn).MaxTextSize = 15
            Win:RegisterText(plusBtn)

            local function formatVal(v)
                local numStr = (stepVal < 1) and string.format("%.2f", v) or tostring(math.floor(v + 0.5))
                return numStr .. suffix
            end

            local function setVal(newVal, fireCb)
                local snapped = math.floor((newVal - minVal) / stepVal + 0.5) * stepVal + minVal
                curVal = math.clamp(snapped, minVal, maxVal)
                if flag then Win.Flags[flag] = curVal end
                local alpha = math.clamp((curVal - minVal) / math.max(0.0001, maxVal - minVal), 0, 1)
                sliderFill.Size = UDim2.new(alpha, 0, 1, 0)
                dragBtn.Position = UDim2.new(alpha, 0, 0.5, 0)
                numBox.Text = formatVal(curVal)
                if fireCb and sCfg.Callback then
                    sCfg.Callback(curVal)
                end
            end

            setVal(curVal, false)

            local sliding = false
            local function updateFromInput(inputPos)
                local absX = sliderHandle.AbsolutePosition.X
                local absW = math.max(1, sliderHandle.AbsoluteSize.X)
                local rel = math.clamp((inputPos.X - absX) / absW, 0, 1)
                local raw = minVal + rel * (maxVal - minVal)
                setVal(raw, true)
            end

            local function beginSlide(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = true
                    Win:PlaySound(true)
                    updateFromInput(input.Position)
                end

            sliderHandle.InputBegan:Connect(beginSlide)
            dragBtn.InputBegan:Connect(beginSlide)

                if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    updateFromInput(input.Position)
                end
            end))

            local endConn = Win:BindConnection(Lib.UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = false
                end
            end))

            slot.Destroying:Connect(function()
                pcall(function() moveConn:Disconnect() end)
                pcall(function() endConn:Disconnect() end)
            end)

            minusBtn.MouseButton1Click:Connect(function()
                Win:PlaySound(true)
                setVal(curVal - stepVal, true)
            end)

            plusBtn.MouseButton1Click:Connect(function()
                Win:PlaySound(true)
                setVal(curVal + stepVal, true)
            end)

            numBox.FocusLost:Connect(function()
                local cleanNum = tonumber(string.match(numBox.Text or "", "[%-]?%d+%.?%d*"))
                if cleanNum then
                    Win:PlaySound(true)
                    setVal(cleanNum, true)
                else
                    numBox.Text = formatVal(curVal)
                end
            end)

            table.insert(Win.RegisteredControlSyncs, function()
                if not sliding and row.Parent then
                    local v = sCfg.Getter and sCfg.Getter() or (flag and Win.Flags[flag]) or curVal
                    setVal(v, false)
                end
            end)

            return {
                Frame = slot,
                Set = function(_, v, fireCb) setVal(v, fireCb) end,
                Get = function() return curVal end,
            }
        end

                        oBtn.Position = UDim2.new(0, 32, 0, 0)
                        task.delay(math.min((i - 1) * 0.02, 0.18), function()
                            if oBtn and oBtn.Parent then
                                Win:Tween(oBtn, { Position = UDim2.new(0, 0, 0, 0) }, 0.22, Enum.EasingStyle.Back)

                        end)

            local function rebuildOptions(newOptions)
                optionsList = newOptions or optionsList
                table.clear(optSlots)
                for _, ch in ipairs(optDrawer:GetChildren()) do
                    if ch:IsA("Frame") then ch:Destroy() end
                end

                            optBtn.Text = (on and "✓ " or "") .. tostring(opt)
                            optBtn.BackgroundColor3 = on and S.AccentColor or Color3.fromRGB(48, 50, 58)
                        else
                            local on = (selectedValue == opt)
                            optBtn.Text = tostring(opt)
                            optBtn.BackgroundColor3 = on and S.AccentColor or Color3.fromRGB(48, 50, 58)
                        end
                    end
                    updateOptVisual()

                    optBtn.MouseButton1Click:Connect(function()
                        Win:PlaySound(true)
                        if isMulti then
                            if type(selectedValue) ~= "table" then selectedValue = {} end
                            selectedValue[opt] = not selectedValue[opt]
                            if flag then Win.Flags[flag] = selectedValue end
                            updateOptVisual()
                            dropHandle.Text = formatDisplay()
                            if callback then callback(selectedValue) end
                        else
                            selectedValue = opt
                            if flag then Win.Flags[flag] = selectedValue end
                            for _, sibBtn in ipairs(optSlots) do
                                sibBtn.BackgroundColor3 = (sibBtn == optBtn) and S.AccentColor or Color3.fromRGB(48, 50, 58)
                            end
                            setDrawerState(false)
                            if callback then callback(opt) end
                        end
                    end)
                end
                dropHandle.Text = formatDisplay()
            end

            rebuildOptions(optionsList)

            local function toggleDrawer()
                Win:PlaySound(false)
                setDrawerState(not drawerOpen)
            end

            dropHandle.MouseButton1Click:Connect(toggleDrawer)

                    selectedValue = (getterFn and getterFn()) or (flag and Win.Flags[flag]) or selectedValue

                end
            end)

                Set = function(_, val, fireCb)
                    selectedValue = val
                    if flag then Win.Flags[flag] = val end
                    dropHandle.Text = formatDisplay()
                    if fireCb and callback then callback(val) end
                end,
            }

    -- =================================================================================
    -- PHẦN 6: HỆ THỐNG LƯU / NẠP CONFIG & XUẤT DỮ LIỆU JSON
    -- =================================================================================
    function Win:GetConfigPath(name)
        local safe = string.gsub(tostring(name or "MacDinh"), "[^%w%._%-]", "_")
        return Win.ConfigFolder .. "/Configs/" .. safe .. ".json"
    end

    function Win:ExportConfigJSON()
        local payload = {
            Settings = Lib.SerializeValue(Win.Settings),
            Flags = Lib.SerializeValue(Win.Flags),
        }
        local ok, str = pcall(function() return Lib.HttpService:JSONEncode(payload) end)
        return ok and str or "{}"
    end

    function Win:SaveConfig(name, silent)
        if type(writefile) ~= "function" then return false end
        Lib.EnsureFolders(Win.ConfigFolder)
        name = (name and name ~= "") and name or Win.SelectedConfigName or "Mặc Định"
        Win.SelectedConfigName = name
        if not table.find(Win.SavedConfigsList, name) then
            table.insert(Win.SavedConfigsList, name)
        end
        local jsonStr = Win:ExportConfigJSON()
        pcall(function()
            writefile(Win:GetConfigPath(name), jsonStr)
            writefile(Win.ConfigFolder .. "/LastSession.json", jsonStr)
        end)
        if not silent then
            Win:Notify({ Title = "LƯU CONFIG", Description = "Đã lưu cấu hình: " .. tostring(name), Duration = 2.5 })
        end
        return true
    end

    function Win:LoadConfig(name, silent)
        if type(isfile) ~= "function" or type(readfile) ~= "function" then return false end
        local path = (name == "__LAST__") and (Win.ConfigFolder .. "/LastSession.json") or Win:GetConfigPath(name)
        local ok, decoded = pcall(function()
            if isfile(path) then
                return Lib.DeserializeValue(Lib.HttpService:JSONDecode(readfile(path)))
            end
            return nil
        end)
        if ok and type(decoded) == "table" then
            if type(decoded.Settings) == "table" then
                for k, v in pairs(decoded.Settings) do
                    if Win.Settings[k] ~= nil then Win.Settings[k] = v end
                end
            end
            if type(decoded.Flags) == "table" then
                for k, v in pairs(decoded.Flags) do
                    Win.Flags[k] = v
                end
            end
            Win:RefreshThemeAndLayout()
            if Win.CurrentActiveTabObj and Win.CurrentActiveTabObj.SyncAll then
                Win.CurrentActiveTabObj:SyncAll()
            end
            if not silent then
                Win:Notify({ Title = "NẠP CONFIG", Description = "Đã nạp cấu hình: " .. tostring(name), Duration = 2.5 })
            end
            return true
        end
        return false
    end

    -- Đóng/Mở Menu kèm hiệu ứng trượt toàn bộ nút
    function Win:SetMenuOpen(openState)
        Win:PlaySound(false)
        Win.MenuOpened = openState
        U.DarkOverlay.Visible = S.DarkOverlayOnOpen and openState
        if U.MenuBlur then
            U.MenuBlur.Enabled = S.WorldBlurOnOpen and openState
            U.MenuBlur.Size = (S.WorldBlurOnOpen and openState) and S.WorldBlurSize or 0
        end

        if openState then
            U.MainImage.Visible = true
            local targetPos = Win.SavedMenuPos or UDim2.new(0.5, 0, 0.5, 0)
            U.MainImage.Position = UDim2.new(targetPos.X.Scale, targetPos.X.Offset, targetPos.Y.Scale + 0.05, targetPos.Y.Offset)
            U.MainImage.Size = UDim2.new(S.WidthScale * 0.75, 0, S.HeightScale * 0.75, 0)
            Win:Tween(U.MainImage, {
                Position = targetPos,
                Size = UDim2.new(S.WidthScale, 0, S.HeightScale, 0),
                BackgroundTransparency = S.GlassTransparency,
            }, 0.28, Enum.EasingStyle.Back)

            Win:PlayNavButtonsSlide(true)
            if U.DetailView.Visible then
                Win:PlayButtonsSlideIn(U.DetailScroll)
            elseif U.GridView.Visible then
                Win:PlayButtonsSlideIn(U.GridScroll)
            elseif U.HomeView.Visible then
                Win:PlayButtonsSlideIn(U.HomeView)
            end
        else
            Win:PlayNavButtonsSlide(false)
            local curPos = U.MainImage.Position
            Win:Tween(U.MainImage, {
                Position = UDim2.new(curPos.X.Scale, curPos.X.Offset, curPos.Y.Scale + 0.04, curPos.Y.Offset),
                Size = UDim2.new(0.14, 0, 0.14, 0),
                BackgroundTransparency = 1,
            }, 0.20, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            task.delay(0.21, function()
                if not Win.MenuOpened and U.MainImage then
                    U.MainImage.Visible = false
                end
            end)
        end
    end

table.insert(Lib.Connections, Lib.RunService.RenderStepped:Connect(function(dt)
    Lib.FPS = math.floor((1 / math.max(dt, 0.001)) * 0.2 + Lib.FPS * 0.8)
end))

table.insert(Lib.Connections, Lib.RunService.Heartbeat:Connect(function()
    local runStr = Lib.FormatHMS(tick() - Lib.StartTick)
    local srvStr = Lib.FormatHMS(Lib.Workspace.DistributedGameTime or 0)
    for _, w in ipairs(Lib.Windows) do
        local U = w.UI
        if U and U.RuntimeLabel and U.RuntimeLabel.Parent then
            U.RuntimeLabel.Text = string.format("@Aris_UILib • FPS: %d • Run: %s • Srv: %s", Lib.FPS, runStr, srvStr)
        end
        if U and U.ChangelogText and U.HomeView and U.HomeView.Visible then
            if w.HomeSummaryGetter then
                U.ChangelogText.Text = tostring(w.HomeSummaryGetter(w))
            else
                U.ChangelogText.Text = w:DefaultHomeSummary()
            end
        end
    end
end))

function Lib.Cleanup()
    for _, conn in ipairs(Lib.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    table.clear(Lib.Connections)
    Lib.DestroyAllExistingMenus()
end
