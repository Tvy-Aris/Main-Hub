local function getSafeParent()
    if type(gethui) == "function" then
        local ok, ui = pcall(gethui)
        if ok and ui then return ui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then return cg end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function make(className, props, children)
    local inst = Instance.new(className)
    if props then
        for k, v in pairs(props) do
            if k ~= "Parent" then
                inst[k] = v
            end
        end
    end
    if children then
        for _, c in ipairs(children) do
            c.Parent = inst
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function clamp(x, lo, hi)
    if x < lo then return lo end
    if x > hi then return hi end
    return x
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

local function tween(obj, duration, props, style, dir)
    local t = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local Maid = {}
Maid.__index = Maid

function Maid.new()
    return setmetatable({ _items = {} }, Maid)
end

function Maid:Give(item)
    table.insert(self._items, item)
    return item
end

function Maid:Clean()
    for i = #self._items, 1, -1 do
        local item = self._items[i]
        self._items[i] = nil
        local t = typeof(item)
        if t == "RBXScriptConnection" then
            item:Disconnect()
        elseif t == "Instance" then
            item:Destroy()
        elseif t == "function" then
            pcall(item)
        end
    end
end

local Spring = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping)
    return setmetatable({
        Position = initial or 0,
        Target = initial or 0,
        Velocity = 0,
        Stiffness = stiffness or 240,
        Damping = damping or 24,
    }, Spring)
end

function Spring:Step(dt)
    local d = math.min(dt, 1 / 30)
    local force = (self.Target - self.Position) * self.Stiffness
    self.Velocity = self.Velocity + (force - self.Velocity * self.Damping) * d
    self.Position = self.Position + self.Velocity * d
    return self.Position
end

-- 3D Wind-Blown Silk Cloth Engine (ViewportFrame + Continuous WedgePart Mesh)
-- Renders a real 3D billowing cloth mesh with anisotropic crimson-black silk

local SilkCloth = {}
SilkCloth.__index = SilkCloth

local COLS = 9
local ROWS = 6
local SPAN_X = 11.4
local SPAN_Y = 7.8
local MAX_PULSES = 3

local LIGHT_KEY = Vector3.new(-0.42, 0.56, 0.71).Unit
local LIGHT_RIM = Vector3.new(0.68, -0.34, 0.65).Unit
local VIEW_DIR  = Vector3.new(0, 0, 1)
local HALF_KEY  = (LIGHT_KEY + VIEW_DIR).Unit
local HALF_RIM  = (LIGHT_RIM + VIEW_DIR).Unit

-- Anisotropic crimson-black silk BRDF evaluated per-triangle normal
local function shadeSilk(normal, height, touchGlow)
    local nDotL = normal:Dot(LIGHT_KEY)
    local wrap = clamp((nDotL + 0.32) / 1.32, 0, 1)

    local nDotH = math.max(0, normal:Dot(HALF_KEY))
    local satinSheen = math.pow(nDotH, 7) * 0.55
    local silkCrest  = math.pow(nDotH, 24) * 0.42

    local rimDotH = math.max(0, normal:Dot(HALF_RIM))
    local rimSheen = math.pow(rimDotH, 10) * 0.24

    local fresnel = math.pow(1 - clamp(math.abs(normal.Z), 0, 1), 2.1) * 0.42
    local foldDepth = clamp((height + 1.1) / 2.2, 0, 1)

    local lum = clamp(wrap * 0.58 + foldDepth * 0.24 + fresnel + touchGlow * 0.16, 0, 1)
    local spec = clamp(satinSheen + silkCrest + rimSheen + touchGlow * 0.14, 0, 1)

    -- Deep obsidian troughs -> oxblood velvet -> rich crimson -> rose-silk glint
    local r = clamp(0.03 + lum * 0.52 + spec * 0.46, 0, 0.96)
    local g = clamp(0.012 + lum * 0.045 + math.pow(spec, 1.8) * 0.36, 0, 0.52)
    local b = clamp(0.02 + lum * 0.085 + math.pow(spec, 1.6) * 0.40, 0, 0.58)
    return Color3.new(r, g, b)
end

-- Exact right-triangle decomposition of an arbitrary 3D triangle (a, b, c)
local function updateTriangle(w1, w2, a, b, c, color)
    local ab = (b - a).Magnitude
    local ac = (c - a).Magnitude
    local bc = (c - b).Magnitude

    if ac > ab and ac > bc then
        b, c = c, b
        ab = ac
    elseif bc > ab and bc > ac then
        a, c = c, a
        ab = bc
    end

    if ab < 1e-4 then return end
    local dir = (b - a) / ab
    local acVec = c - a
    local proj = acVec:Dot(dir)
    local d = a + dir * proj
    local alt = c - d
    local h = alt.Magnitude
    if h < 1e-4 then return end

    local up = alt / h
    local back = dir:Cross(up)
    if back.Magnitude < 1e-4 then return end
    back = back.Unit

    local len1 = math.max(proj, 0.01)
    local len2 = math.max(ab - proj, 0.01)

    w1.Size = Vector3.new(0.08, h, len1)
    w1.CFrame = CFrame.fromMatrix((a + c) * 0.5, -back, up, dir)
    w1.Color = color

    w2.Size = Vector3.new(0.08, h, len2)
    w2.CFrame = CFrame.fromMatrix((b + c) * 0.5, back, up, -dir)
    w2.Color = color
end

function SilkCloth.new(hostFrame, maid, cornerRadius)
    local self = setmetatable({
        Host = hostFrame,
        Maid = maid,
        Time = 0,
        Active = true,
        PointerTarget = Vector2.new(0, 0),
        PointerCurrent = Vector2.new(0, 0),
        PointerVelocity = Vector2.new(0, 0),
        PointerEnergy = 0,
        PointerInside = false,
        PointerDown = false,
        Pulses = {},
        PulseCursor = 1,
        Vertices = {},
        Triangles = {},
    }, SilkCloth)

    for i = 1, MAX_PULSES do
        self.Pulses[i] = { x = 0, y = 0, age = 10, strength = 0 }
    end

    local viewport = make("ViewportFrame", {
        Name = "SilkViewport",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BorderSizePixel = 0,
        Ambient = Color3.new(1, 1, 1),
        LightColor = Color3.fromRGB(40, 20, 26),
        LightDirection = Vector3.new(-0.3, -0.5, -0.8),
        ZIndex = 1,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 16) }),
    })

    local cam = make("Camera", {
        FieldOfView = 34,
        CFrame = CFrame.lookAt(Vector3.new(0, 0, 10.6), Vector3.new(0, 0, 0)),
        Parent = viewport,
    })
    viewport.CurrentCamera = cam

    local meshFolder = make("Folder", { Name = "ClothMesh", Parent = viewport })

    -- Pre-allocate vertex table ((COLS+1) x (ROWS+1))
    local vCount = (COLS + 1) * (ROWS + 1)
    for i = 1, vCount do
        self.Vertices[i] = Vector3.zero
    end

    -- Pre-allocate WedgePart pairs for 2 triangles per quad
    local function newWedge()
        return make("WedgePart", {
            Anchored = true,
            CanCollide = false,
            CastShadow = false,
            Material = Enum.Material.SmoothPlastic,
            Color = Palette.Base,
            Size = Vector3.new(0.08, 0.1, 0.1),
            Parent = meshFolder,
        })
    end

    for row = 0, ROWS - 1 do
        for col = 0, COLS - 1 do
            local i00 = row * (COLS + 1) + col + 1
            local i10 = i00 + 1
            local i01 = i00 + (COLS + 1)
            local i11 = i01 + 1

            table.insert(self.Triangles, { i00, i10, i11, newWedge(), newWedge() })
            table.insert(self.Triangles, { i00, i11, i01, newWedge(), newWedge() })
        end
    end

    -- Soft dark edge vignette so foreground typography stays crisp
    make("Frame", {
        Name = "SilkScrim",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Palette.Base,
        BackgroundTransparency = 0.34,
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 16) }),
        make("UIGradient", {
            Rotation = 90,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 0.30),
                NumberSequenceKeypoint.new(0.45, 0.56),
                NumberSequenceKeypoint.new(1.00, 0.22),
            }),
        }),
    })

    self:_bindInput()
    self:_startLoop()
    return self
end

function SilkCloth:_screenToCloth(screenPos)
    local absPos = self.Host.AbsolutePosition
    local absSize = self.Host.AbsoluteSize
    if absSize.X <= 1 or absSize.Y <= 1 then
        return 0, 0, false
    end
    local nx = ((screenPos.X - absPos.X) / absSize.X) * 2 - 1
    local ny = 1 - ((screenPos.Y - absPos.Y) / absSize.Y) * 2
    local inside = nx >= -1 and nx <= 1 and ny >= -1 and ny <= 1
    return clamp(nx, -1, 1) * (SPAN_X * 0.42), clamp(ny, -1, 1) * (SPAN_Y * 0.42), inside
end

function SilkCloth:TriggerPulse(screenPos, strength)
    local cx, cy = 0, 0
    if screenPos then
        cx, cy = self:_screenToCloth(screenPos)
    end
    local slot = self.Pulses[self.PulseCursor]
    slot.x = cx
    slot.y = cy
    slot.age = 0
    slot.strength = strength or 1.0
    self.PulseCursor = (self.PulseCursor % MAX_PULSES) + 1
end

function SilkCloth:_bindInput()
    self.Maid:Give(UserInputService.InputChanged:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            self.PointerInside = inside
            if inside then
                self.PointerTarget = Vector2.new(cx, cy)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputBegan:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            if inside then
                self.PointerInside = true
                self.PointerDown = true
                self.PointerTarget = Vector2.new(cx, cy)
                self:TriggerPulse(input.Position, 1.0)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            self.PointerDown = false
        end
    end))
end

function SilkCloth:_startLoop()
    self.Maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not self.Active or not self.Host.Visible then return end
        local step = math.min(dt, 0.05)
        self.Time = self.Time + step
        local t = self.Time

        -- Smoothly track pointer position and drag velocity
        local prevPtr = self.PointerCurrent
        local rate = 1 - math.exp(-step * 10)
        self.PointerCurrent = self.PointerCurrent:Lerp(self.PointerTarget, rate)
        local vel = (self.PointerCurrent - prevPtr) / math.max(step, 1 / 120)
        self.PointerVelocity = self.PointerVelocity:Lerp(vel, 1 - math.exp(-step * 8))

        local targetEnergy = self.PointerDown and 1.25 or (self.PointerInside and 0.72 or 0)
        self.PointerEnergy = lerp(self.PointerEnergy, targetEnergy, 1 - math.exp(-step * 7))

        -- Advance ripple pulses
        for i = 1, MAX_PULSES do
            local p = self.Pulses[i]
            if p.strength > 0 then
                p.age = p.age + step
                if p.age > 1.8 then
                    p.strength = 0
                end
            end
        end

        local px, py = self.PointerCurrent.X, self.PointerCurrent.Y
        local dragSpeed = math.min(self.PointerVelocity.Magnitude * 0.04, 0.65)
        local energy = self.PointerEnergy

        -- 1. Evaluate wind-blown silk cloth heightfield at every grid vertex
        local verts = self.Vertices
        local idx = 1
        for row = 0, ROWS do
            local vFrac = row / ROWS
            local baseY = (vFrac - 0.5) * SPAN_Y
            for col = 0, COLS do
                local uFrac = col / COLS
                local baseX = (uFrac - 0.5) * SPAN_X

                -- Multi-octave travelling wind waves (diagonal gust + soft silk flutter)
                local w1 = math.sin(baseX * 0.52 + baseY * 0.36 - t * 1.18) * 0.62
                local w2 = math.sin(baseX * 0.94 - baseY * 0.62 - t * 1.72 + w1 * 0.95) * 0.30
                local w3 = math.cos(baseX * 1.55 + baseY * 1.12 - t * 2.35 + w2 * 0.65) * 0.12

                -- Subtle lateral wind sway
                local swayX = math.sin(baseY * 0.55 - t * 0.95) * 0.16
                local swayY = math.cos(baseX * 0.45 - t * 0.80) * 0.12

                local z = w1 + w2 + w3

                -- Interactive finger/cursor cloth press + raised crease collar
                if energy > 0.01 then
                    local dx = baseX - px
                    local dy = baseY - py
                    local r2 = dx * dx + dy * dy
                    local dent = -math.exp(-r2 / 2.2) * (0.68 + dragSpeed * 0.45) * energy
                    local r = math.sqrt(r2)
                    local collar = math.exp(-math.pow(r - 1.9, 2) / 1.1) * 0.22 * energy
                    z = z + dent + collar
                end

                -- Expanding Gaussian-cosine cloth ripples from taps/clicks
                for pIdx = 1, MAX_PULSES do
                    local pulse = self.Pulses[pIdx]
                    if pulse.strength > 0 then
                        local dx = baseX - pulse.x
                        local dy = baseY - pulse.y
                        local dist = math.sqrt(dx * dx + dy * dy)
                        local front = (dist - pulse.age * 5.8) / 1.15
                        local env = math.exp(-(front * front)) * pulse.strength * math.exp(-pulse.age * 1.65)
                        z = z + math.cos(front * 2.4) * env * 0.52
                    end
                end

                verts[idx] = Vector3.new(baseX + swayX, baseY + swayY, z)
                idx = idx + 1
            end
        end

        -- 2. Shade and update all 3D silk cloth triangles
        for _, tri in ipairs(self.Triangles) do
            local a = verts[tri[1]]
            local b = verts[tri[2]]
            local c = verts[tri[3]]

            local normal = (b - a):Cross(c - a)
            local nLen = normal.Magnitude
            if nLen > 1e-5 then
                normal = normal / nLen
                if normal.Z < 0 then
                    normal = -normal
                end
            else
                normal = VIEW_DIR
            end

            local centroid = (a + b + c) / 3
            local touchGlow = 0
            if energy > 0.01 then
                local dx = centroid.X - px
                local dy = centroid.Y - py
                touchGlow = math.exp(-(dx * dx + dy * dy) / 3.8) * energy
            end

            local color = shadeSilk(normal, centroid.Z, touchGlow)
            updateTriangle(tri[4], tri[5], a, b, c, color)
        end
    end))
end

local Velour = {
    Version = "2.0.0",
    Flags = {},
}
Velour.__index = Velour

local function attachDrag(handle, target, onTap)
    local dragging = false
    local dragStart, startPos
    local moved = false
    local conns = {}

    table.insert(conns, handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))

    table.insert(conns, UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                moved = true
            end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end
    end))

    table.insert(conns, UserInputService.InputEnded:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            if not moved and onTap then
                onTap(input.Position)
            end
        end
    end))

    return function()
        for _, c in ipairs(conns) do
            c:Disconnect()
        end
    end
end

        if self.Destroyed then return end

            tween(main, 0.24, { Size = finalSize })
            silk:TriggerPulse(main.AbsolutePosition + main.AbsoluteSize * 0.5, 0.85)

            silk.Active = false
            local tw = tween(main, 0.16, { Size = UDim2.fromOffset(targetW * 0.94, targetH * 0.94) })
            tw.Completed:Once(function()
                if not self.Visible and not self.Destroyed then

                    floatBtn.Visible = true

            end)
        end

        end

                t.Page.Visible = false

            end

            tab.Page.Visible = true
            tab.Page.Position = UDim2.fromOffset(0, 6)

            selectTab()
            silk:TriggerPulse(btn.AbsolutePosition + btn.AbsoluteSize * 0.5, 0.75)
        end)

        table.insert(self.Tabs, tab)
        if #self.Tabs == 1 then
            selectTab()

            function section:AddSlider(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Slider"
                local minV = cfg.Min or 0
                local maxV = cfg.Max or 100
                local step = cfg.Increment or 1
                local suffix = cfg.Suffix or ""
                local flag = cfg.Flag
                local value = clamp(cfg.Default or minV, minV, maxV)
                local callback = cfg.Callback

                local row = addRow(54)

                make("TextLabel", {

                    AnchorPoint = Vector2.new(1, 0),
                    Position = UDim2.new(1, -14, 0, 9),
                    Size = UDim2.fromOffset(66, 16),
                    BackgroundTransparency = 1,
                    Text = tostring(value) .. suffix,
                    TextColor3 = Palette.TextSecondary,
                    TextSize = 11,
                    FontFace = Fonts.Mono,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    ZIndex = 18,
                    Parent = row,
                })

                local hit = make("TextButton", {
                    Position = UDim2.new(0, 14, 0, 26),
                    Size = UDim2.new(1, -28, 0, 22),
                    BackgroundTransparency = 1,
                    Text = "",
                    AutoButtonColor = false,
                    ZIndex = 19,
                    Parent = row,
                })

                local rail = make("Frame", {
                    AnchorPoint = Vector2.new(0, 0.5),
                    Position = UDim2.fromScale(0, 0.5),
                    Size = UDim2.new(1, 0, 0, 3),
                    BackgroundColor3 = Palette.Hairline,
                    BackgroundTransparency = 0.86,
                    BorderSizePixel = 0,
                    ZIndex = 18,
                    Parent = hit,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local fill = make("Frame", {
                    Size = UDim2.fromScale(0, 1),
                    BackgroundColor3 = Palette.Accent,
                    BorderSizePixel = 0,
                    ZIndex = 19,
                    Parent = rail,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local thumb = make("Frame", {
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.fromScale(1, 0.5),
                    Size = UDim2.fromOffset(10, 10),
                    BackgroundColor3 = Palette.TextPrimary,
                    BorderSizePixel = 0,
                    ZIndex = 20,
                    Parent = fill,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(1, 0) }),
                })

                local obj = {}
                local dragging = false

                local function fmt(n)
                    if step < 1 then
                        local d = math.max(1, math.ceil(-math.log10(step)))
                        return string.format("%." .. d .. "f", n)
                    end
                    return tostring(math.floor(n + 0.5))
                end

                function obj:Set(v, silent)
                    local snapped = math.floor(((v - minV) / step) + 0.5) * step + minV
                    value = clamp(snapped, minV, maxV)
                    if flag then Velour.Flags[flag] = value end
                    local pct = (maxV == minV) and 0 or (value - minV) / (maxV - minV)
                    readout.Text = fmt(value) .. suffix
                    tween(fill, 0.06, { Size = UDim2.fromScale(pct, 1) })
                    if not silent and callback then
                        task.spawn(callback, value)
                    end
                end

                function obj:Get()
                    return value
                end

                local function updateX(x)
                    local rel = (x - rail.AbsolutePosition.X) / math.max(rail.AbsoluteSize.X, 1)
                    obj:Set(minV + clamp(rel, 0, 1) * (maxV - minV))
                end

                hit.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = true
                        tween(thumb, 0.12, { Size = UDim2.fromOffset(13, 13) })
                        tween(readout, 0.12, { TextColor3 = Palette.AccentBright })
                        updateX(input.Position.X)
                        silk:TriggerPulse(input.Position, 0.6)
                    end
                end)

                maid:Give(UserInputService.InputChanged:Connect(function(input)
                    if not dragging then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                        updateX(input.Position.X)
                    end
                end))

                maid:Give(UserInputService.InputEnded:Connect(function(input)
                    if not dragging then return end
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = false
                        tween(thumb, 0.12, { Size = UDim2.fromOffset(10, 10) })
                        tween(readout, 0.12, { TextColor3 = Palette.TextSecondary })
                    end
                end))

                obj:Set(value, true)
                return obj
            end

                    tween(row, 0.14, { BackgroundTransparency = 1 })
                    tween(arrow, 0.14, { TextColor3 = Palette.TextMuted, Position = UDim2.new(1, -14, 0.5, 0) })
                end)

                btn.MouseButton1Click:Connect(function()
                    silk:TriggerPulse(row.AbsolutePosition + row.AbsoluteSize * 0.5, 0.9)
                    if callback then task.spawn(callback) end
                end)

                return row
            end

            --------------------------------------------------------------------
            -- Dropdown (Single & Multi)
            --------------------------------------------------------------------

            function section:AddDropdown(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Select"
                local items = cfg.Values or {}
                local multi = cfg.Multi == true
                local flag = cfg.Flag
                local callback = cfg.Callback

                local selected = multi and {} or cfg.Default
                if multi and type(cfg.Default) == "table" then
                    for _, v in ipairs(cfg.Default) do selected[v] = true end
                end

                local row = addRow(40)
                row.ClipsDescendants = true

                local topBtn = make("TextButton", {
                    Size = UDim2.new(1, 0, 0, 40),
                    BackgroundTransparency = 1,
                    Text = "",
                    AutoButtonColor = false,
                    ZIndex = 18,
                    Parent = row,
                })

                make("TextLabel", {
                    Position = UDim2.new(0, 14, 0, 0),
                    Size = UDim2.new(0.48, -14, 0, 40),
                    BackgroundTransparency = 1,
                    Text = label,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 12,
                    FontFace = Fonts.Body,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = topBtn,
                })

                local valLbl = make("TextLabel", {
                    AnchorPoint = Vector2.new(1, 0),
                    Position = UDim2.new(1, -28, 0, 0),
                    Size = UDim2.new(0.48, -14, 0, 40),
                    BackgroundTransparency = 1,
                    Text = "None",
                    TextColor3 = Palette.TextSecondary,
                    TextSize = 11,
                    FontFace = Fonts.Reg,
                    TextTruncate = Enum.TextTruncate.AtEnd,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    ZIndex = 18,
                    Parent = topBtn,
                })

                local chev = make("TextLabel", {
                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -12, 0.5, 0),
                    Size = UDim2.fromOffset(14, 14),
                    BackgroundTransparency = 1,
                    Text = "v",
                    TextColor3 = Palette.TextMuted,
                    TextSize = 9,
                    FontFace = Fonts.Mono,
                    ZIndex = 18,
                    Parent = topBtn,
                })

                local list = make("Frame", {
                    Position = UDim2.fromOffset(10, 42),
                    Size = UDim2.new(1, -20, 0, 0),
                    AutomaticSize = Enum.AutomaticSize.Y,
                    BackgroundTransparency = 1,
                    ZIndex = 18,
                    Parent = row,
                }, {
                    make("UIListLayout", { Padding = UDim.new(0, 3) }),
                    make("UIPadding", { PaddingBottom = UDim.new(0, 8) }),
                })

                local open = false
                local obj = {}

                local function summarize()
                    if multi then
                        local out = {}
                        for _, item in ipairs(items) do
                            if selected[item] then table.insert(out, tostring(item)) end
                        end
                        valLbl.Text = (#out > 0) and table.concat(out, ", ") or "None"
                    else
                        valLbl.Text = selected ~= nil and tostring(selected) or "None"
                    end
                    if flag then Velour.Flags[flag] = selected end
                end

                local function setOpen(state)
                    open = state
                    local h = open and (50 + #items * 29) or 40
                    tween(row, 0.2, { Size = UDim2.new(1, 0, 0, h) })
                    tween(chev, 0.18, { Rotation = open and 180 or 0 })
                end

                function obj:Refresh(newItems)
                    items = newItems or {}
                    for _, c in ipairs(list:GetChildren()) do
                        if c:IsA("TextButton") then c:Destroy() end
                    end
                    for _, item in ipairs(items) do
                        local isSel = multi and selected[item] or (selected == item)
                        local opt = make("TextButton", {
                            Size = UDim2.new(1, 0, 0, 26),
                            BackgroundColor3 = isSel and Palette.AccentSoft or Palette.Hairline,
                            BackgroundTransparency = isSel and 0.35 or 0.94,
                            Text = "  " .. tostring(item),
                            TextColor3 = isSel and Palette.TextPrimary or Palette.TextSecondary,
                            TextSize = 11,
                            FontFace = Fonts.Body,
                            TextXAlignment = Enum.TextXAlignment.Left,
                            AutoButtonColor = false,
                            ZIndex = 19,
                            Parent = list,
                        }, {
                            make("UICorner", { CornerRadius = UDim.new(0, 6) }),
                        })

                        opt.MouseButton1Click:Connect(function()
                            if multi then
                                selected[item] = not selected[item]
                                local active = selected[item]
                                tween(opt, 0.12, {
                                    BackgroundColor3 = active and Palette.AccentSoft or Palette.Hairline,
                                    BackgroundTransparency = active and 0.35 or 0.94,
                                    TextColor3 = active and Palette.TextPrimary or Palette.TextSecondary,
                                })
                                summarize()
                                if callback then task.spawn(callback, selected) end
                            else
                                selected = item
                                summarize()
                                obj:Refresh(items)
                                setOpen(false)
                                if callback then task.spawn(callback, selected) end
                            end
                            silk:TriggerPulse(opt.AbsolutePosition + opt.AbsoluteSize * 0.5, 0.6)
                        end)
                    end
                    if open then setOpen(true) end
                end

                function obj:Set(val, silent)
                    selected = val
                    summarize()
                    obj:Refresh(items)
                    if not silent and callback then task.spawn(callback, selected) end
                end

                function obj:Get()
                    return selected
                end

                topBtn.MouseButton1Click:Connect(function()
                    setOpen(not open)
                    silk:TriggerPulse(topBtn.AbsolutePosition + topBtn.AbsoluteSize * 0.5, 0.6)
                end)

                summarize()
                obj:Refresh(items)
                return obj
            end

            function section:AddInput(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Input"
                local placeholder = cfg.Placeholder or "Enter value..."
                local numeric = cfg.Numeric == true
                local flag = cfg.Flag
                local callback = cfg.Callback

                local row = addRow(42)

                make("TextLabel", {
                    Position = UDim2.new(0, 14, 0, 0),
                    Size = UDim2.new(0.48, -14, 1, 0),
                    BackgroundTransparency = 1,
                    Text = label,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 12,
                    FontFace = Fonts.Body,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = row,
                })

                local box = make("TextBox", {
                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -12, 0.5, 0),
                    Size = UDim2.new(0.46, 0, 0, 26),
                    BackgroundColor3 = Palette.Base,
                    BackgroundTransparency = 0.35,
                    Text = cfg.Default or "",
                    PlaceholderText = placeholder,
                    PlaceholderColor3 = Palette.TextMuted,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 11,
                    FontFace = Fonts.Mono,
                    ClearTextOnFocus = false,
                    ZIndex = 19,
                    Parent = row,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(0, 6) }),
                    make("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8) }),
                })
                local bStroke = make("UIStroke", {
                    Color = Palette.Hairline,
                    Thickness = 1,
                    Transparency = 0.86,
                    Parent = box,
                })

                box.Focused:Connect(function()
                    tween(bStroke, 0.14, { Color = Palette.Accent, Transparency = 0.3 })
                end)

                box.FocusLost:Connect(function(enter)
                    tween(bStroke, 0.14, { Color = Palette.Hairline, Transparency = 0.86 })
                    local val = box.Text
                    if numeric then
                        val = tonumber(val) or 0
                        box.Text = tostring(val)
                    end
                    if flag then Velour.Flags[flag] = val end
                    if callback then task.spawn(callback, val, enter) end
                end)

                return {
                    Set = function(_, v)
                        box.Text = tostring(v)
                        if flag then Velour.Flags[flag] = v end
                    end,
                    Get = function()
                        return numeric and (tonumber(box.Text) or 0) or box.Text
                    end,
                }
            end

            function section:AddKeybind(cfg)
                cfg = cfg or {}
                local label = cfg.Title or "Keybind"
                local key = cfg.Default or Enum.KeyCode.F
                local flag = cfg.Flag
                local callback = cfg.Callback

                local row = addRow(40)

                make("TextLabel", {
                    Position = UDim2.new(0, 14, 0, 0),

                    BackgroundTransparency = 1,
                    Text = label,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 12,
                    FontFace = Fonts.Body,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = row,
                })

                local keyBtn = make("TextButton", {
                    AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -12, 0.5, 0),
                    Size = UDim2.fromOffset(72, 24),
                    BackgroundColor3 = Palette.Hairline,
                    BackgroundTransparency = 0.90,
                    Text = key.Name,
                    TextColor3 = Palette.TextPrimary,
                    TextSize = 11,
                    FontFace = Fonts.Mono,
                    AutoButtonColor = false,
                    ZIndex = 19,
                    Parent = row,
                }, {
                    make("UICorner", { CornerRadius = UDim.new(0, 6) }),
                })

                local listening = false
                if flag then Velour.Flags[flag] = key end

                keyBtn.MouseButton1Click:Connect(function()
                    listening = true
                    keyBtn.Text = "..."
                    silk:TriggerPulse(keyBtn.AbsolutePosition + keyBtn.AbsoluteSize * 0.5, 0.6)
                end)

                maid:Give(UserInputService.InputBegan:Connect(function(input, gp)
                    if listening then
                        if input.UserInputType == Enum.UserInputType.Keyboard then
                            listening = false
                            key = input.KeyCode
                            keyBtn.Text = key.Name
                            if flag then Velour.Flags[flag] = key end
                        end
                    elseif not gp and input.KeyCode == key then
                        if callback then task.spawn(callback, key) end
                    end
                end))

                return {
                    Set = function(_, k)
                        key = k
                        keyBtn.Text = key.Name
                        if flag then Velour.Flags[flag] = key end
                    end,
                    Get = function() return key end,
                }
            end

        return tab
    end

end

--------------------------------------------------------------------------------
-- Config Export
--------------------------------------------------------------------------------

function Velour:ExportConfig()
    local out = {}
    for k, v in pairs(self.Flags) do
        local t = typeof(v)
        if t == "boolean" or t == "number" or t == "string" or t == "table" then
            out[k] = v
        elseif t == "EnumItem" then
            out[k] = v.Name
        end
    end
    return HttpService:JSONEncode(out)
end

return Velour
