local Env = (type(getgenv) == "function" and getgenv()) or _G

local function cleanupPreviousInstance()
    if type(Env.__VELOUR_INSTANCE__) == "table" then
        local prev = Env.__VELOUR_INSTANCE__
        if type(prev.UserCleanups) == "table" then
            for _, fn in ipairs(prev.UserCleanups) do
                pcall(fn)
            end
        end
        if type(prev.Destroy) == "function" then
            pcall(prev.Destroy)
        end
        Env.__VELOUR_INSTANCE__ = nil
    end
    if type(Env._VelourActiveDestroy) == "function" then
        pcall(Env._VelourActiveDestroy)
        Env._VelourActiveDestroy = nil
    end
    local function purge(parent)
        if not parent then return end
        for _, child in ipairs(parent:GetChildren()) do
            if child.Name == "VelourScreenGui" then
                pcall(function() child:Destroy() end)
            end
        end
    end
    if type(gethui) == "function" then
        local ok, hui = pcall(gethui)
        if ok then purge(hui) end
    end
    pcall(function() purge(CoreGui) end)
    pcall(function() purge(LocalPlayer:FindFirstChild("PlayerGui")) end)
end

local function getSafeParent()
    if type(gethui) == "function" then
        local ok, ui = pcall(gethui)
        if ok and ui then return ui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then return cg end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function make(className, props, children)
    local inst = Instance.new(className)
    if props then
        for k, v in pairs(props) do
            if k ~= "Parent" then
                inst[k] = v
            end
        end
    end
    if children then
        for _, c in ipairs(children) do
            c.Parent = inst
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function clamp(x, lo, hi)
    if x < lo then return lo end
    if x > hi then return hi end
    return x
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

local function darkenColor(c, factor)
    return Color3.new(c.R * factor, c.G * factor, c.B * factor)
end

local function colorToTable(c)
    return { math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5) }
end

local function tableToColor(t)
    if type(t) == "table" and #t == 3 then
        return Color3.fromRGB(t[1], t[2], t[3])
    end
    return nil
end

local function tween(obj, duration, props, style, dir)
    local t = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local function attachCleanPress(button, targetFrame, flashFrame, themeRef, scaleAmount)
    local uiScale = targetFrame:FindFirstChildOfClass("UIScale") or make("UIScale", {
        Scale = 1,
        Parent = targetFrame,
    })
    local pressScale = scaleAmount or 0.98

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(uiScale, 0.08, { Scale = pressScale }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            if flashFrame then
                flashFrame.BackgroundColor3 = darkenColor(themeRef.Accent, 0.42)
                tween(flashFrame, 0.08, { BackgroundTransparency = 0.58 })
            end
        end
    end)

    local function release()
        tween(uiScale, 0.20, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        if flashFrame then
            tween(flashFrame, 0.22, { BackgroundTransparency = 1 })
        end
    end

    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            release()
        end
    end)

    button.MouseLeave:Connect(release)
end

function Maid.new()
    return setmetatable({ _items = {} }, Maid)
end

function Maid:Give(item)
    table.insert(self._items, item)
    return item
end

function Maid:Clean()
    for i = #self._items, 1, -1 do
        local item = self._items[i]
        self._items[i] = nil
        local t = typeof(item)
        if t == "RBXScriptConnection" then
            item:Disconnect()
        elseif t == "Instance" then
            item:Destroy()
        elseif t == "function" then
            pcall(item)
        end
    end
end

local Spring = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping)
    return setmetatable({
        Position = initial or 0,
        Target = initial or 0,
        Velocity = 0,
        Stiffness = stiffness or 290,
        Damping = damping or 22,
    }, Spring)
end

function Spring:Step(dt)
    local d = math.min(dt, 1 / 30)
    local force = (self.Target - self.Position) * self.Stiffness
    self.Velocity = self.Velocity + (force - self.Velocity * self.Damping) * d
    self.Position = self.Position + self.Velocity * d
    return self.Position
end

-- 3D Reflective Polygonal Silk Engine (-20% Frequency: 16x10 = 320 Triangles)
-- Fixes black spots during ripples via dual-hemisphere velvet wrap lighting &
-- safe Z-amplitude clamping, plus live Character Movement & Camera Rotation
-- reactive cloth drift and specular shimmer!
--------------------------------------------------------------------------------

local SilkCloth = {}
SilkCloth.__index = SilkCloth

local COLS = 16
local ROWS = 10
local SPAN_X = 14.4
local SPAN_Y = 9.8
local MAX_PULSES = 10

local BASE_LIGHT_KEY = Vector3.new(-0.42, 0.56, 0.71).Unit
local LIGHT_RIM      = Vector3.new(0.68, -0.34, 0.65).Unit
local VIEW_DIR       = Vector3.new(0, 0, 1)
local HALF_RIM       = (LIGHT_RIM + VIEW_DIR).Unit

-- Dual-hemisphere silk BRDF: guarantees zero pitch-black holes on steep ripples
local function shadeSilk(normal, height, touchGlow, theme, lightKey, halfKey)
    local nDotL = normal:Dot(lightKey)
    -- Smooth hemisphere wrap (0.20 floor prevents dead black cutouts in steep troughs)
    local wrap = 0.22 + 0.78 * (0.5 + 0.5 * nDotL)
    local rimFill = 0.5 + 0.5 * normal:Dot(LIGHT_RIM)

    local nDotH = math.max(0, normal:Dot(halfKey))
    local satinSheen = math.pow(nDotH, 5.2) * 0.58
    local sharpGlint = math.pow(nDotH, 16) * 0.50

    local rimDotH = math.max(0, normal:Dot(HALF_RIM))
    local rimSheen = math.pow(rimDotH, 7.5) * 0.28

    local fresnel = math.pow(1 - clamp(math.abs(normal.Z), 0, 1), 1.8) * 0.36
    local foldDepth = clamp((height + 1.1) / 2.2, 0.14, 1)

    local bodyMix = clamp(wrap * 0.56 + rimFill * 0.16 + foldDepth * 0.22 + fresnel * 0.55 + touchGlow * 0.22, 0.20, 1)

    local sh = theme.SilkShadow
    local md = theme.SilkMid
    local pr = theme.SilkPrimary
    local hi = theme.SilkHighlight
    local gl = theme.SilkGlint
    local rm = theme.SilkRim
    local tg = theme.SilkTouchGlow

    -- 3-stop body: Shadow -> Mid -> Main
    local bodyCurve = math.pow(bodyMix, 1.12)
    local r, g, b
    if bodyCurve < 0.5 then
        local k = bodyCurve * 2
        r = lerp(sh.R, md.R, k); g = lerp(sh.G, md.G, k); b = lerp(sh.B, md.B, k)
    else
        local k = (bodyCurve - 0.5) * 2
        r = lerp(md.R, pr.R, k); g = lerp(md.G, pr.G, k); b = lerp(md.B, pr.B, k)
    end

    -- Specular layers, each with its own color
    local wSat  = clamp((satinSheen + touchGlow * 0.30) * 0.82, 0, 1)
    local wGlnt = clamp(sharpGlint * 0.82, 0, 1)
    local wRim  = clamp(rimSheen * 0.82, 0, 1)
    local total = wSat + wGlnt + wRim
    if total > 1e-4 then
        local mixAmt = math.min(total, 1)
        local sr = (hi.R * wSat + gl.R * wGlnt + rm.R * wRim) / total
        local sg = (hi.G * wSat + gl.G * wGlnt + rm.G * wRim) / total
        local sb = (hi.B * wSat + gl.B * wGlnt + rm.B * wRim) / total
        r = r + (sr - r) * mixAmt
        g = g + (sg - g) * mixAmt
        b = b + (sb - b) * mixAmt
    end

    -- Touch glow tint
    if touchGlow > 0.01 then
        local k = clamp(touchGlow * 0.28, 0, 0.5)
        r = r + (tg.R - r) * k
        g = g + (tg.G - g) * k
        b = b + (tg.B - b) * k
    end

    return Color3.new(clamp(r, 0, 1), clamp(g, 0, 1), clamp(b, 0, 1))
end

local function updateTriangle(w1, w2, a, b, c, color)
    local ab = (b - a).Magnitude
    local ac = (c - a).Magnitude
    local bc = (c - b).Magnitude

    if ac > ab and ac > bc then
        b, c = c, b
        ab = ac
    elseif bc > ab and bc > ac then
        a, c = c, a
        ab = bc
    end

    if ab < 1e-4 then return end
    local dir = (b - a) / ab
    local acVec = c - a
    local proj = acVec:Dot(dir)
    local d = a + dir * proj
    local alt = c - d
    local h = alt.Magnitude
    if h < 1e-4 then return end

    local up = alt / h
    local back = dir:Cross(up)
    if back.Magnitude < 1e-4 then return end
    back = back.Unit

    local len1 = math.max(proj, 0.01) + 0.042
    local len2 = math.max(ab - proj, 0.01) + 0.042
    local hh = h + 0.042

    w1.Size = Vector3.new(0.18, hh, len1)
    w1.CFrame = CFrame.fromMatrix((a + c) * 0.5, -back, up, dir)
    w1.Color = color

    w2.Size = Vector3.new(0.18, hh, len2)
    w2.CFrame = CFrame.fromMatrix((b + c) * 0.5, back, up, -dir)
    w2.Color = color
end

function SilkCloth.new(hostFrame, maid, themeRef, cornerRadius)
    local self = setmetatable({
        Host = hostFrame,
        Maid = maid,
        Theme = themeRef,
        Time = 0,
        Active = true,
        PointerTarget = Vector2.new(0, 0),
        PointerCurrent = Vector2.new(0, 0),
        PointerVelocity = Vector2.new(0, 0),
        PointerEnergy = 0,
        PointerInside = false,
        PointerDown = false,
        Pulses = {},
        Vertices = {},
        Triangles = {},
        -- Character movement & camera rotation reactive state
        DriftOffset = Vector2.new(0, 0),
        CamTilt = Vector2.new(0, 0),
        LastCamLook = nil,
    }, SilkCloth)

    for i = 1, MAX_PULSES do
        self.Pulses[i] = { x = 0, y = 0, age = 10, strength = 0 }
    end

    -- Background matches deep darkened primary instead of pure black (prevents dark bleed)
    local viewport = make("ViewportFrame", {
        Name = "SilkViewport",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = themeRef.SilkBackdrop,
        BorderSizePixel = 0,
        Ambient = Color3.new(1, 1, 1),
        LightColor = Color3.new(0, 0, 0),
        LightDirection = Vector3.new(0, 0, -1),
        ZIndex = 1,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
    })
    self.Viewport = viewport

    -- FOV 38 + 16x10 grid = ~20% lower polygon frequency vs 6.0, clean & fast
    local cam = make("Camera", {
        FieldOfView = 38,
        CFrame = CFrame.lookAt(Vector3.new(0, 0, 10.2), Vector3.new(0, 0, 0)),
        Parent = viewport,
    })
    viewport.CurrentCamera = cam

    local meshFolder = make("Folder", { Name = "ClothMesh", Parent = viewport })

    local vCount = (COLS + 1) * (ROWS + 1)
    for i = 1, vCount do
        self.Vertices[i] = Vector3.zero
    end

    local function newWedge()
        return make("WedgePart", {
            Anchored = true,
            CanCollide = false,
            CastShadow = false,
            Material = Enum.Material.SmoothPlastic,
            Color = darkenColor(themeRef.SilkPrimary, 0.2),
            Size = Vector3.new(0.18, 0.1, 0.1),
            Parent = meshFolder,
        })
    end

    for row = 0, ROWS - 1 do
        for col = 0, COLS - 1 do
            local i00 = row * (COLS + 1) + col + 1
            local i10 = i00 + 1
            local i01 = i00 + (COLS + 1)
            local i11 = i01 + 1

            table.insert(self.Triangles, { i00, i10, i11, newWedge(), newWedge() })
            table.insert(self.Triangles, { i00, i11, i01, newWedge(), newWedge() })
        end
    end

    local scrim = make("Frame", {
        Name = "SilkScrim",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = themeRef.SilkScrim,
        BackgroundTransparency = 0.28,
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
        make("UIGradient", {
            Rotation = 90,
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0.00, 0.26),
                NumberSequenceKeypoint.new(0.45, 0.56),
                NumberSequenceKeypoint.new(1.00, 0.20),
            }),
        }),
    })
    self.Scrim = scrim

    -- Opening veil: covers the viewport's first blank frames with a silk tone and
    -- fades out, so the menu never flashes black while the mesh warms up
    self.Veil = make("Frame", {
        Name = "SilkOpenVeil",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = themeRef.SilkMid,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = 3,
        Parent = hostFrame,
    }, {
        make("UICorner", { CornerRadius = UDim.new(0, cornerRadius or 24) }),
    })

    self:_bindInput()
    self:_startLoop()
    return self
end

function SilkCloth:_screenToCloth(screenPos)
    local absPos = self.Host.AbsolutePosition
    local absSize = self.Host.AbsoluteSize
    if absSize.X <= 1 or absSize.Y <= 1 then
        return 0, 0, false
    end
    local nx = ((screenPos.X - absPos.X) / absSize.X) * 2 - 1
    local ny = 1 - ((screenPos.Y - absPos.Y) / absSize.Y) * 2
    local inside = nx >= -1 and nx <= 1 and ny >= -1 and ny <= 1
    return clamp(nx, -1, 1) * 5.2, clamp(ny, -1, 1) * 3.4, inside
end

function SilkCloth:TriggerPulse(screenPos, strength)
    local cx, cy = 0, 0
    if screenPos then
        cx, cy = self:_screenToCloth(screenPos)
    end

    local chosen = self.Pulses[1]
    for i = 1, MAX_PULSES do
        local slot = self.Pulses[i]
        if slot.strength <= 0 then
            chosen = slot
            break
        elseif slot.age > chosen.age then
            chosen = slot
        end
    end

    chosen.x = cx
    chosen.y = cy
    chosen.age = 0
    chosen.strength = strength or 1.0
end

function SilkCloth:_bindInput()
    self.Maid:Give(UserInputService.InputChanged:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            self.PointerInside = inside
            if inside then
                self.PointerTarget = Vector2.new(cx, cy)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputBegan:Connect(function(input)
        if not self.Active or not self.Host.Visible then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local cx, cy, inside = self:_screenToCloth(input.Position)
            if inside then
                self.PointerInside = true
                self.PointerDown = true
                self.PointerTarget = Vector2.new(cx, cy)
                self:TriggerPulse(input.Position, 1.05)
            end
        end
    end))

    self.Maid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            self.PointerDown = false
        end
    end))
end

function SilkCloth:FadeIn(duration)
    local veil = self.Veil
    if not veil or not veil.Parent then return end
    veil.BackgroundColor3 = self.Theme.SilkMid
    veil.BackgroundTransparency = 0
    tween(veil, duration or 0.5, { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
end

function SilkCloth:Refresh()
    self:_update(1 / 60)
end

function SilkCloth:_update(dt)
        if not self.Active or not self.Host.Visible then return end
        local step = math.min(dt, 0.05)
        self.Time = self.Time + step
        local t = self.Time
        local theme = self.Theme

        self.Viewport.BackgroundColor3 = theme.SilkBackdrop
        self.Scrim.BackgroundColor3 = theme.SilkScrim

        -- Sample character movement & camera rotation for live parallax motion
        local worldCam = workspace.CurrentCamera
        local moveX, moveY = 0, 0
        local yawDelta, pitchDelta = 0, 0

        if worldCam then
            local camCF = worldCam.CFrame
            local look = camCF.LookVector
            if self.LastCamLook then
                local deltaLook = look - self.LastCamLook
                yawDelta = clamp(deltaLook:Dot(camCF.RightVector) * 18, -1.6, 1.6)
                pitchDelta = clamp(deltaLook:Dot(camCF.UpVector) * 18, -1.6, 1.6)
            end
            self.LastCamLook = look

            local char = LocalPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                local localVel = camCF:VectorToObjectSpace(root.AssemblyLinearVelocity)
                moveX = clamp(localVel.X * 0.045, -1.8, 1.8)
                moveY = clamp((-localVel.Z + localVel.Y * 0.5) * 0.045, -1.8, 1.8)
            end
        end

        -- Accumulate smooth drift from movement & spring-damped tilt from camera spin
        self.DriftOffset = self.DriftOffset + Vector2.new(moveX + yawDelta * 1.4, moveY + pitchDelta * 1.2) * step * 1.65
        self.CamTilt = self.CamTilt:Lerp(Vector2.new(yawDelta * 0.55 + moveX * 0.25, pitchDelta * 0.55 + moveY * 0.20), 1 - math.exp(-step * 8))

        local lightKey = Vector3.new(
            BASE_LIGHT_KEY.X + clamp(self.CamTilt.X * 0.55, -0.45, 0.45),
            BASE_LIGHT_KEY.Y + clamp(self.CamTilt.Y * 0.55, -0.45, 0.45),
            BASE_LIGHT_KEY.Z
        ).Unit
        local halfKey = (lightKey + VIEW_DIR).Unit

        local prevPtr = self.PointerCurrent
        self.PointerCurrent = self.PointerCurrent:Lerp(self.PointerTarget, 1 - math.exp(-step * 18))
        local vel = (self.PointerCurrent - prevPtr) / math.max(step, 1 / 120)
        self.PointerVelocity = self.PointerVelocity:Lerp(vel, 1 - math.exp(-step * 10))

        local targetEnergy = self.PointerDown and 1.35 or (self.PointerInside and 0.85 or 0)
        self.PointerEnergy = lerp(self.PointerEnergy, targetEnergy, 1 - math.exp(-step * 9))

        for i = 1, MAX_PULSES do
            local p = self.Pulses[i]
            if p.strength > 0 then
                p.age = p.age + step
                if p.age > 2.2 then p.strength = 0 end
            end
        end

        local px, py = self.PointerCurrent.X, self.PointerCurrent.Y
        local dragBoost = math.min(self.PointerVelocity.Magnitude * 0.05, 0.65)
        local energy = self.PointerEnergy
        local driftX, driftY = self.DriftOffset.X, self.DriftOffset.Y

        -- 1. Evaluate 16x10 cloth vertices with movement/camera parallax & safe wave clamp
        local verts = self.Vertices
        local idx = 1
        for row = 0, ROWS do
            local baseY = (row / ROWS - 0.5) * SPAN_Y
            for col = 0, COLS do
                local baseX = (col / COLS - 0.5) * SPAN_X

                local sx = baseX + driftX
                local sy = baseY + driftY

                local w1 = math.sin(sx * 0.52 + sy * 0.36 - t * 1.18) * 0.50
                local w2 = math.sin(sx * 0.96 - sy * 0.62 - t * 1.72 + w1 * 0.82) * 0.24
                local w3 = math.cos(sx * 1.62 + sy * 1.22 - t * 2.34 + w2 * 0.55) * 0.10

                local swayX = math.sin(sy * 0.48 - t * 0.90) * 0.11
                local swayY = math.cos(sx * 0.40 - t * 0.76) * 0.08

                local z = w1 + w2 + w3

                if energy > 0.01 then
                    local dx = baseX - px
                    local dy = baseY - py
                    local r2 = dx * dx + dy * dy
                    local dent = -math.exp(-r2 / 1.85) * (0.64 + dragBoost * 0.36) * energy
                    local r = math.sqrt(r2)
                    local collar = math.exp(-math.pow(r - 1.65, 2) / 0.95) * 0.26 * energy
                    z = z + dent + collar
                end

                local waveSum = 0
                for pIdx = 1, MAX_PULSES do
                    local pulse = self.Pulses[pIdx]
                    if pulse.strength > 0 then
                        local dx = baseX - pulse.x
                        local dy = baseY - pulse.y
                        local dist = math.sqrt(dx * dx + dy * dy)
                        local front = (dist - pulse.age * 5.2) / 0.96
                        local env = math.exp(-(front * front)) * pulse.strength * math.exp(-pulse.age * 1.50)
                        waveSum = waveSum + math.sin(front * 2.9) * env * 0.52
                    end
                end
                z = clamp(z + clamp(waveSum, -0.82, 0.82), -1.35, 1.35)

                verts[idx] = Vector3.new(baseX + swayX, baseY + swayY, z)
                idx = idx + 1
            end
        end

        -- 2. Shade and update all 320 reflective triangles
        for _, tri in ipairs(self.Triangles) do
            local a = verts[tri[1]]
            local b = verts[tri[2]]
            local c = verts[tri[3]]

            local normal = (b - a):Cross(c - a)
            local nLen = normal.Magnitude
            if nLen > 1e-5 then
                normal = normal / nLen
                if normal.Z < 0 then normal = -normal end
            else
                normal = VIEW_DIR
            end

            local centroid = (a + b + c) / 3
            local touchGlow = 0
            if energy > 0.01 then
                local dx = centroid.X - px
                local dy = centroid.Y - py
                touchGlow = math.exp(-(dx * dx + dy * dy) / 2.9) * energy
            end

            local color = shadeSilk(normal, centroid.Z, touchGlow, theme, lightKey, halfKey)
            updateTriangle(tri[4], tri[5], a, b, c, color)
        end
end

local function attachIrregularBorder(targetFrame, maid, themeRef, thickness)
    local stroke = make("UIStroke", {
        Color = Color3.new(1, 1, 1),
        Thickness = thickness or 1.8,
        Transparency = 0.05,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = targetFrame,
    })

    local grad = make("UIGradient", {
        Rotation = 0,
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, themeRef.SilkShadow),
            ColorSequenceKeypoint.new(0.50, themeRef.BorderGradient),
            ColorSequenceKeypoint.new(1.00, themeRef.SilkShadow),
        }),
        Parent = stroke,
    })

    local phase = math.random() * 12
    local angle = math.random(0, 360)
    local surge = 0

    maid:Give(RunService.RenderStepped:Connect(function(dt)
        if not targetFrame.Visible then return end
        local step = math.min(dt, 0.05)
        phase = phase + step
        surge = surge * math.exp(-step * 4.0)

        local chaoticSpeed = 24
            + 62 * math.sin(phase * 1.4142) * math.cos(phase * 0.6180)
            + 44 * math.sin(phase * 2.7183 + math.cos(phase * 1.732))
            + surge * 280

        angle = (angle + chaoticSpeed * step) % 360
        grad.Rotation = angle

        local baseDark = themeRef.SilkShadow
        local midTone  = darkenColor(themeRef.BorderGradient, 0.45)
        local peakTone = themeRef.BorderGradient

        local p1 = clamp(0.22 + 0.13 * math.sin(phase * 1.618 + 0.7), 0.08, 0.34)
        local p2 = clamp(0.50 + 0.14 * math.cos(phase * 2.236 - 0.4), p1 + 0.08, 0.68)
        local p3 = clamp(p2 + 0.14, p2 + 0.06, 0.90)

        grad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, baseDark),
            ColorSequenceKeypoint.new(p1, midTone),
            ColorSequenceKeypoint.new(p2, peakTone),
            ColorSequenceKeypoint.new(p3, midTone),
            ColorSequenceKeypoint.new(1.00, baseDark),
        })
    end))

    return function(amount)
        surge = math.min(surge + (amount or 1.0), 2.4)
    end
end

local function attachDrag(handle, target, onTap)
    local dragging = false
    local dragStart, startPos
    local moved = false
    local conns = {}

    table.insert(conns, handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))

    table.insert(conns, UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                moved = true
            end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end
    end))

    table.insert(conns, UserInputService.InputEnded:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            if not moved and onTap then
                onTap(input.Position)
            end
        end
    end))

    return function()
        for _, c in ipairs(conns) do
            c:Disconnect()
        end
    end
end

function Velour:RegisterCleanup(fn)
    if type(fn) ~= "function" then return end
    if type(Env.__VELOUR_INSTANCE__) == "table" then
        table.insert(Env.__VELOUR_INSTANCE__.UserCleanups, fn)
    end
end

    local function getToggleIconCenter()
        local pos = toggleSquare.AbsolutePosition
        local sz = toggleSquare.AbsoluteSize
        return UDim2.fromOffset(pos.X + sz.X * 0.5, pos.Y + sz.Y * 0.5)

    function window:SetVisible(state)
        if self.Destroyed then return end
        self.Visible = state
        local iconOrigin = getToggleIconCenter()

        if state then
            main.Position = iconOrigin
            mainScale.Scale = 0.08
            main.Visible = true
            silk.Active = true
            silk:Refresh()      -- build mesh colors before the first visible frame
            silk:FadeIn(0.5)    -- soft silk-colored veil instead of black

            -- Restore every tab's scroll position (layout needs a frame or two to settle)
            task.spawn(function()
                for _ = 1, 3 do
                    RunService.RenderStepped:Wait()
                    if window.Destroyed or not window.Visible then return end
                    for _, tb in ipairs(window.Tabs) do
                        tb.UpdateCanvas()
                        local maxY = math.max(0, tb.Page.CanvasSize.Y.Offset - tb.Page.AbsoluteWindowSize.Y)
                        tb.Page.CanvasPosition = Vector2.new(tb.ScrollPos.X, math.min(tb.ScrollPos.Y, maxY))
                    end
                end
            end)
            tween(main, 0.34, { Position = self._savedOpenPos }, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
            tween(mainScale, 0.36, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            silk:TriggerPulse(main.AbsolutePosition + main.AbsoluteSize * 0.5, 1.05)
        else
            self._savedOpenPos = main.Position
            for _, tb in ipairs(self.Tabs) do
                tb.ScrollPos = tb.Page.CanvasPosition
            end
            silk.Active = false
            tween(main, 0.24, { Position = iconOrigin }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
            local tw = tween(mainScale, 0.24, { Scale = 0.06 }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
            tw.Completed:Once(function()
                if not self.Visible and not self.Destroyed then
                    main.Visible = false
                    tween(toggleScale, 0.10, { Scale = 1.14 }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
                    task.delay(0.10, function()
                        if toggleScale and toggleScale.Parent then
                            tween(toggleScale, 0.20, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        end
                    end)
                end
            end)
        end
    end

    function window:ResetConfig()
        if fsAvailable and type(delfile) == "function" then
            pcall(function()
                if isfile(configPath) then delfile(configPath) end
            end)
        end
        for k, v in pairs(DefaultTheme) do
            theme[k] = v
        end
        broadcastTheme()
        applyToggleIcon(options.ToggleIcon or "")
        for flag, entry in pairs(controlsRegistry) do
            if entry.Set then
                pcall(entry.Set, entry.Obj, entry.Default, false)
            end
        end
        saveConfigNow()
        self:Notify({
            Title = "Config Reset",
            Content = "All settings and theme colors restored to defaults.",
            Duration = 2.6,
        })
    end

    function window:Destroy()
        if self.Destroyed then return end
        self.Destroyed = true
        silk.Active = false
        maid:Clean()
        if Env.__VELOUR_INSTANCE__ and Env.__VELOUR_INSTANCE__.Destroy == self._destroyRef then
            Env.__VELOUR_INSTANCE__ = nil
        end
    end

        })

        local btnScale = make("UIScale", { Scale = 1, Parent = btn })

        local page = make("ScrollingFrame", {
            Name = "Page_" .. tabTitle,
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ScrollBarThickness = 3,
            ScrollBarImageColor3 = theme.Accent,
            ScrollBarImageTransparency = 0.20,
            CanvasSize = UDim2.fromOffset(0, 0),
            AutomaticCanvasSize = Enum.AutomaticSize.None,
            ElasticBehavior = Enum.ElasticBehavior.WhenScrollable,
            ScrollingEnabled = true,
            Visible = false,
            ZIndex = 15,
            Parent = pageHolder,
        }, {
            make("UIPadding", {
                PaddingTop = UDim.new(0, 4),
                PaddingBottom = UDim.new(0, 28),
                PaddingRight = UDim.new(0, 8),
            }),
        })

        local pageLayout = make("UIListLayout", {
            Padding = UDim.new(0, 14),
            SortOrder = Enum.SortOrder.LayoutOrder,
            Parent = page,
        })

        local function updateCanvas()
            -- While the window is hidden the layout reports a bogus size and Roblox
            -- clamps CanvasPosition to 0 -> that was the "jump back to top" bug
            if not main.Visible or not window.Visible then return end
            local contentY = pageLayout.AbsoluteContentSize.Y
            if contentY <= 0 then return end
            page.CanvasSize = UDim2.fromOffset(0, contentY + 42)
        end
        maid:Give(pageLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(updateCanvas))

        onThemeChanged(function(t)
            page.ScrollBarImageColor3 = t.Accent
            if window.ActiveTab and window.ActiveTab.Button == btn then
                btn.TextColor3 = t.TextPrimary
            else
                btn.TextColor3 = t.TextSecondary
            end
        end)

        local tab = {
            Index = tabIndex,
            Title = tabTitle,
            Button = btn,
            Page = page,
            Order = 0,
            UpdateCanvas = updateCanvas,
            ScrollPos = Vector2.new(0, 0),
        }

        local function selectTab()
            if self.ActiveTab == tab then return end
            local prevIndex = self.ActiveTab and self.ActiveTab.Index or 0
            local dir = (tab.Index >= prevIndex) and 1 or -1

            for _, t in ipairs(self.Tabs) do
                t.Page.Visible = false
                tween(t.Button, 0.18, { TextColor3 = theme.TextSecondary })
            end

            self.ActiveTab = tab
            tab.Page.Visible = true
            updateCanvas()
            tween(tab.Button, 0.18, { TextColor3 = theme.TextPrimary })

            if prevIndex > 0 then
                tab.Page.Position = UDim2.fromOffset(dir * 44, 0)
                tween(tab.Page, 0.34, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            else
                tab.Page.Position = UDim2.fromOffset(0, 0)
            end
        end

        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(btnScale, 0.08, { Scale = 0.90 })
                spawnTabWaterWave(dockWaveLayer, input.Position, theme)
                pulseDockBorder(1.3)
            end
        end)

        btn.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(btnScale, 0.22, { Scale = 1 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            end
        end)

        btn.MouseButton1Click:Connect(function()
            selectTab()
            silk:TriggerPulse(btn.AbsolutePosition + btn.AbsoluteSize * 0.5, 0.95)
        end)

        table.insert(self.Tabs, tab)
        task.defer(syncDockWidth)

        if #self.Tabs == 1 then
            selectTab()
        end

                    if step < 1 then

                    end
                    return tostring(math.floor(n + 0.5))

                function obj:Get() return value end

                local function updateX(x)
                    local rel = (x - rail.AbsolutePosition.X) / math.max(rail.AbsoluteSize.X, 1)
                    obj:Set(minV + clamp(rel, 0, 1) * (maxV - minV))
                end

                local function endDrag()
                    if window._activeSlider ~= obj then return end
                    window._activeSlider = nil
                    activeInput = nil
                    page.ScrollingEnabled = true
                    tween(thumb, 0.14, { Size = UDim2.fromOffset(13, 13) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                    tween(readout, 0.14, { TextColor3 = theme.TextSecondary })
                    tween(readoutScale, 0.14, { Scale = 1 })
                end

                hit.InputBegan:Connect(function(input)
                    if window._activeSlider ~= nil then return end
                    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                        window._activeSlider = obj
                        activeInput = input
                        page.ScrollingEnabled = false
                        tween(thumb, 0.12, { Size = UDim2.fromOffset(16, 16) }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
                        tween(readout, 0.12, { TextColor3 = theme.SilkHighlight })
                        tween(readoutScale, 0.12, { Scale = 1.07 })
                        updateX(input.Position.X)
                        silk:TriggerPulse(input.Position, 0.75)
                    end
                end)

                maid:Give(UserInputService.InputChanged:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement
                        or (input.UserInputType == Enum.UserInputType.Touch and input == activeInput) then
                        updateX(input.Position.X)
                    end
                end))

                maid:Give(UserInputService.InputEnded:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input == activeInput
                        or (activeInput and activeInput.UserInputType == Enum.UserInputType.MouseButton1
                            and input.UserInputType == Enum.UserInputType.MouseButton1) then
                        endDrag()
                    end
                end))

                if flag then
                    controlsRegistry[flag] = { Obj = obj, Set = obj.Set, Default = defaultVal }
                end

                obj:Set(value, not (flag and savedData.Flags[flag] ~= nil))
                return obj
            end

                        updateHue(input.Position)
                    end
                end)

                maid:Give(UserInputService.InputChanged:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input.UserInputType == Enum.UserInputType.MouseMovement
                        or (input.UserInputType == Enum.UserInputType.Touch and input == activeInput) then
                        if dragMode == "SV" then updateSV(input.Position)
                        elseif dragMode == "H" then updateHue(input.Position) end
                    end
                end))

                maid:Give(UserInputService.InputEnded:Connect(function(input)
                    if window._activeSlider ~= obj then return end
                    if input == activeInput
                        or (activeInput and activeInput.UserInputType == Enum.UserInputType.MouseButton1
                            and input.UserInputType == Enum.UserInputType.MouseButton1) then
                        window._activeSlider = nil
                        dragMode = nil
                        activeInput = nil
                        page.ScrollingEnabled = true
                    end
                end))

                topBtn.MouseButton1Click:Connect(function()
                    open = not open
                    local tw = tween(row, 0.22, { Size = UDim2.new(1, 0, 0, open and 166 or 44) })
                    tw.Completed:Once(updateCanvas)
                    silk:TriggerPulse(swatch.AbsolutePosition + swatch.AbsoluteSize * 0.5, 0.75)
                end)

                if flag then
                    controlsRegistry[flag] = { Obj = obj, Set = obj.Set, Default = defaultCol }
                end
                applyHSV(true)
                return obj
            end

            --------------------------------------------------------------------
            -- Paragraph
            --------------------------------------------------------------------

            function section:AddParagraph(cfg)
                cfg = cfg or {}
                local row = addRow(0)
                row.AutomaticSize = Enum.AutomaticSize.Y

                make("UIPadding", {
                    PaddingTop = UDim.new(0, 11),
                    PaddingBottom = UDim.new(0, 11),
                    PaddingLeft = UDim.new(0, 15),
                    PaddingRight = UDim.new(0, 15),
                    Parent = row,
                })
                make("UIListLayout", { Padding = UDim.new(0, 4), Parent = row })

                local tLbl = make("TextLabel", {
                    Size = UDim2.new(1, 0, 0, 18),
                    BackgroundTransparency = 1,
                    Text = cfg.Title or "",
                    TextColor3 = theme.TextPrimary,
                    TextSize = 13,
                    FontFace = Fonts.Semi,
                    TextTruncate = Enum.TextTruncate.AtEnd,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = row,
                })

                local bLbl = make("TextLabel", {
                    Size = UDim2.new(1, 0, 0, 0),
                    AutomaticSize = Enum.AutomaticSize.Y,
                    BackgroundTransparency = 1,
                    Text = cfg.Content or "",
                    TextColor3 = theme.TextSecondary,
                    TextSize = 11,
                    LineHeight = 1.3,
                    FontFace = Fonts.Reg,
                    TextWrapped = true,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    ZIndex = 18,
                    Parent = row,
                })

                onThemeChanged(function(t)
                    tLbl.TextColor3 = t.TextPrimary
                    bLbl.TextColor3 = t.TextSecondary
                end)

                return {
                    SetTitle = function(_, s) tLbl.Text = s end,
                    SetContent = function(_, s) bLbl.Text = s end,
                }
            end

            return section
        end

        return tab
    end

    ----------------------------------------------------------------------------
    -- Built-In Deep Theme & Config Studio Tab
    ----------------------------------------------------------------------------

    if includeThemeTab then
        task.defer(function()
            if window.Destroyed then return end
            local themeTab = window:AddTab({ Title = "Theme", LayoutOrder = 999 })

            local presetSec = themeTab:AddSection("Theme Presets & Icon")
            local colorPickers = {}

            local presetNames = {}
            for name in pairs(ThemePresets) do
                table.insert(presetNames, name)
            end
            table.sort(presetNames)

            presetSec:AddDropdown({
                Title = "Preset",
                Values = presetNames,
                Default = "Crimson Obsidian",
                Callback = function(name)
                    local p = ThemePresets[name]
                    if not p then return end
                    for k, c in pairs(p) do
                        theme[k] = c
                        if colorPickers[k] then
                            colorPickers[k]:Set(c, true)
                        end
                    end
                    broadcastTheme()
                    scheduleAutoSave()
                end,
            })

            presetSec:AddInput({
                Title = "Toggle Icon Asset ID",
                Placeholder = "rbxassetid://... or ID",
                Default = toggleIconId,
                Callback = function(val)
                    window:SetToggleIcon(val)
                end,
            })

            local deepSec = themeTab:AddSection("Deep Color Customization")
            local colorFields = {
                { Key = "SilkPrimary",    Label = "Silk Main Color" },
                { Key = "SilkShadow",     Label = "Silk Shadow / Base" },
                { Key = "SilkMid",        Label = "Silk Mid Fold Tone" },
                { Key = "SilkHighlight",  Label = "Silk Specular Reflection" },
                { Key = "SilkGlint",      Label = "Silk Sharp Glint" },
                { Key = "SilkRim",        Label = "Silk Rim Light" },
                { Key = "SilkTouchGlow",  Label = "Silk Touch / Cursor Glow" },
                { Key = "SilkBackdrop",   Label = "Silk Backdrop (behind mesh)" },
                { Key = "SilkScrim",      Label = "Silk Dark Overlay" },
                { Key = "Accent",         Label = "Accent & Controls" },
                { Key = "BorderGradient", Label = "Animated Border Gradient" },
                { Key = "OuterBorder",    Label = "Window Outer Border" },
                { Key = "GlassPanel",     Label = "Glass Panel Surface" },
                { Key = "DockBg",         Label = "Tab Dock & Icon Surface" },
                { Key = "TextPrimary",    Label = "Primary Text" },
                { Key = "TextSecondary",  Label = "Secondary Text" },
            }

            for _, field in ipairs(colorFields) do
                colorPickers[field.Key] = deepSec:AddColorPicker({
                    Title = field.Label,
                    Default = theme[field.Key],
                    Callback = function(c)
                        theme[field.Key] = c
                        broadcastTheme()
                        scheduleAutoSave()
                    end,
                })
            end

            local cfgSec = themeTab:AddSection("Configuration")
            cfgSec:AddButton({
                Title = "Save Config Now",
                Description = "Saves all flags and colors to executor workspace",
                Callback = function()
                    local ok = window:SaveConfig()
                    window:Notify({
                        Title = ok and "Config Saved" or "Workspace Unavailable",
                        Content = ok and ("Saved to " .. configPath) or "Executor does not support writefile.",
                        Duration = 2.5,
                    })
                end,
            })

            cfgSec:AddButton({
                Title = "Reset All Config",
                Description = "Deletes saved file and restores every default",
                Callback = function()
                    window:ResetConfig()
                    for k, picker in pairs(colorPickers) do
                        picker:Set(theme[k], true)
                    end
                end,
            })
        end)
    end

    return window
end

--------------------------------------------------------------------------------
-- Config JSON Export
--------------------------------------------------------------------------------

function Velour:ExportConfig()
    local out = {}
    for k, v in pairs(self.Flags) do
        local t = typeof(v)
        if t == "boolean" or t == "number" or t == "string" or t == "table" then
            out[k] = v
        elseif t == "EnumItem" then
            out[k] = v.Name
        elseif t == "Color3" then
            out[k] = colorToTable(v)
        end
    end
    return HttpService:JSONEncode(out)
end

return Velour
